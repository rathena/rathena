//===== Ragnarok Offline =====================================
//= Clothes Dyer — Ginedin Rephere
//===== Description: =========================================
//= Live cloth-color preview (changelook, not saved) using GRF
//= palette indices for the current job, sex, and Second Costume.
//= Confirm uses setlook and consumes Clothing Dye Coupon (6046)
//= or Clothing Dye Coupon II (6047). prompt() cancel and the
//= client window close both revert the preview.
//= Palettes: npc/custom/clothes_dye_palettes.txt
//============================================================

prt_in,284,168,2	script	Dyer Ginedin Rephere	55,{
	.n$ = "[Dyer Ginedin Rephere]";
	mes .n$;
	mes "11... 12... Mmm... good. I think I'll be able to finish before tonight's party. Oh! I didn't notice you were here!";
	while (1) {
		next;
		mes .n$;
		mes "How may I assist you?";
		next;
		switch (select("- Talk:- Dye Clothing:- Price List:- ^777777Cancel^000000")) {
		case 1:
			mes .n$;
			mes "Life may have gotten a bit better, but when I look at all of the orders I've received... Whew!";
			mes "It seems that the young women of Rune-Midgarts must be very well off these days.";
			next;
			mes .n$;
			mes "Not that I think that they are indulging in anything luxurious, mind you.";
			mes "I don't think there is anything wrong with the pursuit of beauty and being fashionable.";
			next;
			mes .n$;
			mes "For us humans, who have no colorful fur or decorative scales, clothes are one of the few ways we have to display our personality, style, and beauty.";
			next;
			mes .n$;
			mes "Haha... When I think about it, I really feel that my job is worthwhile.  I believe that I provide a service that the people desire.";
			next;
			mes .n$;
			mes "Can you feel it too? The energy that is released when rough fabric and leather are brought to life with color?.....";
			next;
			mes .n$;
			mes "Although the process is very tedious and time consuming, the joy and happiness I feel when a dress is finished more than makes up for all of the hard work!!";
			break;
		case 2:
			callsub S_Dye;
			end;
		case 3:
			mes .n$;
			mes "A new color takes one " + getitemname(.NewCoupon) + ".";
			mes "Back to the original cloth takes one " + getitemname(.OrigCoupon) + ".";
			mes "If you already wear the color you keep, I charge nothing.";
			next;
			callsub S_TicketTalk;
			break;
		case 4:
			mes .n$;
			mes "Make yourself at home. Even though I don't have time for you.";
			close;
		}
	}
	end;

S_TicketTalk:
	mes .n$;
	mes "Those coupons carry a fixer. Without it the color will not stay in the cloth.";
	mes "I do not sell them.";
	next;
	mes .n$;
	mes "I have heard someone does.";
	mes "Where? I have no idea.";
	return;

S_Dye:
	.@orig = getlook(LOOK_CLOTHES_COLOR);
	.@preview = .@orig;
	.@second = 0;
	if (inarray(.SecondIds[0], getlook(LOOK_BODY2)) >= 0)
		.@second = 1;
	.@n = callfunc("F_ClothDyePals", Class, (Sex == SEX_MALE), .@second);
	if (.@n < 1) {
		mes .n$;
		mes "Wow, I've never seen clothes like that before! I'm sorry, but I don't think I can paint it.";
		close;
	}
	mes .n$;
	mes "Oh... you need my work? Well... okay, sounds good.";
	mes "Pick a number. I'll put it on you so you can see it.";
	mes "It will not stay until I set it with a coupon.";
	next;
	while (1) {
		.@menu$ = "";
		for (.@i = 0; .@i < .@n; ++.@i) {
			.@id = @cdye_pal[.@i];
			if (.@id == 0)
				.@label$ = "Original";
			else
				.@label$ = "Color " + .@id;
			if (.@id == .@preview)
				.@label$ = "^0055FF" + .@label$ + "^000000";
			.@menu$ += "- " + .@label$ + ":";
		}
		.@menu$ += "- Confirm this color:- ^777777Cancel^000000";
		.@s = prompt(.@menu$);
		if (.@s == 255 || .@s == .@n + 2) {
			changelook LOOK_CLOTHES_COLOR, .@orig;
			mes .n$;
			mes "As you were.";
			close;
		}
		if (.@s == .@n + 1) {
			if (.@preview == .@orig) {
				changelook LOOK_CLOTHES_COLOR, .@orig;
				mes .n$;
				mes "That is already your color. Nothing to take.";
				close;
			}
			if (.@preview == 0)
				.@need = .OrigCoupon;
			else
				.@need = .NewCoupon;
			if (countitem(.@need) < 1) {
				mes .n$;
				mes "I need a " + getitemname(.@need) + " for that.";
				next;
				callsub S_TicketTalk;
				next;
				continue;
			}
			delitem .@need, 1;
			setlook LOOK_CLOTHES_COLOR, .@preview;
			mes .n$;
			if (.@preview == 0)
				mes "Original cloth. There. It will stay.";
			else
				mes "Color " + .@preview + ". There. It will stay.";
			close;
		}
		.@preview = @cdye_pal[.@s - 1];
		changelook LOOK_CLOTHES_COLOR, .@preview;
	}
	end;

OnInit:
	.NewCoupon = 6046;
	.OrigCoupon = 6047;
	setarray .SecondIds[0],
		Job_Rune_Knight_2nd,
		Job_Mechanic_2nd,
		Job_Guillotine_Cross_2nd,
		Job_Warlock_2nd,
		Job_Archbishop_2nd,
		Job_Ranger_2nd,
		Job_Royal_Guard_2nd,
		Job_Genetic_2nd,
		Job_Shadow_Chaser_2nd,
		Job_Sorcerer_2nd,
		Job_Sura_2nd,
		Job_Minstrel_2nd,
		Job_Wanderer_2nd;
	setunittitle getnpcid(0), "<Clothes Dye>";
	end;
}
