airplane_01,1,1,0	script	International_Airship	-1,{
	end;

OnInit:
	set $@airplanelocation2, -1; // Initial state: in flight
	set $@airplane_status, 0;   // Initial state: in flight
	initnpctimer;
	end;

OnEnable:
	initnpctimer;
	end;

OnTimer25000:
	mapannounce "airplane_01", "This airship is heading to Izlude.", bc_map, "0x00ff00";
	end;

OnTimer50000:
	mapannounce "airplane_01", "We will arrive in Izlude shortly.", bc_map, "0x00ff00";
	end;

OnTimer60000: // Izlude arrival
	set $@airplanelocation2, 1;
	set $@airplane_status, 1; // Docked
	mapannounce "airplane_01", "We have arrived in Izlude. Have a safe trip.", bc_map, "0x00ff00";
	end;

OnTimer70000:
	mapannounce "airplane_01", "We are currently in Izlude. The airship will depart shortly.", bc_map, "0x00ff00";
	end;

OnTimer80000: // Izlude departure
	set $@airplane_status, 0; // In flight
	mapannounce "airplane_01", "The airship is taking off. Our next destination is Juno.", bc_map, "0x70dbdb";
	end;

OnTimer105000:
	mapannounce "airplane_01", "This airship is heading to Juno.", bc_map, "0x70dbdb";
	end;

OnTimer130000:
	mapannounce "airplane_01", "We will arrive in Juno shortly.", bc_map, "0x70dbdb";
	end;

OnTimer140000: // Juno arrival
	set $@airplanelocation2, 2;
	set $@airplane_status, 1; // Docked
	mapannounce "airplane_01", "We have arrived in Juno. Have a safe trip.", bc_map, "0x70dbdb";
	end;

OnTimer150000:
	mapannounce "airplane_01", "We are currently in Juno. The airship will depart shortly.", bc_map, "0x70dbdb";
	end;

OnTimer160000: // Juno departure
	set $@airplane_status, 0; // In flight
	mapannounce "airplane_01", "The airship is taking off. Our next destination is Rachel.", bc_map, "0xFF8200";
	end;

OnTimer185000:
	mapannounce "airplane_01", "This airship is heading to Rachel.", bc_map, "0xFF8200";
	end;

OnTimer210000:
	mapannounce "airplane_01", "We will arrive in Rachel shortly.", bc_map, "0xFF8200";
	end;

OnTimer220000: // Rachel arrival
	set $@airplanelocation2, 0;
	set $@airplane_status, 1; // Docked
	mapannounce "airplane_01", "We have arrived in Rachel. Have a safe trip.", bc_map, "0xFF8200";
	end;

OnTimer230000:
	mapannounce "airplane_01", "We are currently in Rachel. The airship will depart shortly.", bc_map, "0xFF8200";
	end;

OnTimer240000: // Rachel departure
	set $@airplane_status, 0; // In flight
	mapannounce "airplane_01", "The airship is taking off. Our next destination is Izlude.", bc_map, "0x00ff00";
	stopnpctimer;
	// Preserve the monster attack event used by the airship quest.
	.moninv++;
	if (.moninv == 7) {
		if (rand(1,3) == 3) {
			donpcevent "Airship#airplane02::OnEnable";
			end;
		}
		.moninv = 0;
	}
	initnpctimer;
	end;
}

airplane_01,50,66,5	script	Apple Merchant#airplane	86,{
	mes "[Meltz]";
	mes "Welcome! I sell fresh Apples and can turn your Apples into Apple Juice~.";
	next;
	switch (select("Buy Apples:Make Apple Juice:Cancel")) {
	case 1:
		mes "[Meltz]";
		mes "Please enter the number you would like to buy.";
		mes "Apples cost 15 Zeny each. You may buy";
		mes "up to 500 at a time. To cancel,";
		mes "please enter 0.";
		next;
		while (1) {
			input .@input;
			set .@pay, .@input * 15;
			if (.@input == 0) {
				mes "[Meltz]";
				mes "Please come again.";
				mes "Goodbye~.";
				close;
			}
			else if (.@input < 1 || .@input > 500) {
				mes "[Meltz]";
				mes "That amount is outside the allowed range.";
				mes "Please try again.";
				next;
			}
			else {
				mes "[Meltz]";
				mes "You would like to buy ^FF0000" + .@input + " Apples^000000.";
				mes "The total price is ^FF0000" + .@pay + " Zeny^000000.";
				mes "Would you like to make this purchase?";
				next;
				if (select("Yes:No") == 2) {
					mes "[Meltz]";
					mes "Please come again.";
					mes "Goodbye~.";
					close;
				}
				break;
			}
		}
		if (Zeny < .@pay) {
			mes "[Meltz]";
			mes "It looks like you do not have enough Zeny.";
			mes "Please bring more Zeny, or try";
			mes "buying fewer Apples.";
			close;
		}
		else if (checkweight(512,.@input) == 0) {
			mes "[Meltz]";
			mes "You are carrying too many items,";
			mes "or your inventory is too heavy";
			mes "to make this purchase.";
			mes "Please make some room in your inventory";
			mes "and come back.";
			close;
		}
		else {
			set Zeny, Zeny - .@pay;
			getitem 512,.@input; //Apple
			mes "[Meltz]";
			mes "Thank you! If you need Apples,";
			mes "please visit me anytime.";
			close;
		}
	case 2:
		mes "[Meltz]";
		mes "To make Apple Juice, bring me ^FF00003 Apples and 1 Empty Bottle^000000.";
		mes "I will turn them into juice for you.";
		mes "Would you like some Apple Juice?";
		next;
		switch (select("Yes:No")) {
		case 1:
			if (countitem(512) < 3 || countitem(713) < 1) {
				mes "[Meltz]";
				mes "You do not have enough ingredients.";
				mes "Let me remind you. Bring me ^FF00003 Apples";
				mes "and 1 Empty Bottle^000000,";
				mes "and I will make the juice for you.";
				mes "Come see me when you are ready.";
				close;
			}
			else {
				mes "[Meltz]";
				mes "Please wait a moment.";
				next;
				mes "^3355FFGrind, grind,";
				mes "and grind some more~";
				mes "Clink, clink....^000000";
				next;
				// Recheck after the dialogue pauses, before consuming either ingredient.
				if (countitem(512) < 3 || countitem(713) < 1) {
					mes "[Meltz]";
					mes "You no longer have enough ingredients to make Apple Juice.";
					close;
				}
				if (!checkweight(531,1)) {
					mes "[Meltz]";
					mes "Please make room in your inventory for the Apple Juice first.";
					close;
				}
				delitem 512,3; //Apple
				delitem 713,1; //Empty_Bottle
				getitem 531,1; //Apple_Juice
				mes "[Meltz]";
				mes "Here you are~.";
				mes "Have a wonderful time!";
				mes "Whenever you need Apples or Apple Juice,";
				mes "just come and";
				mes "Come see me when you are ready.";
				close;
			}
		case 2:
			mes "[Meltz]";
			mes "Thank you! If you need Apples,";
			mes "please visit me anytime.";
			close;
		}
	case 3:
		mes "[Meltz]";
		mes "Please come again.";
		mes "Goodbye~.";
		close;
	}
}

function	script	applegamble	{
	.@npc_name$ = getarg(0);
