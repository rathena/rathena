//============================================================
// Eden Group Leveling Board (130 - 190)
//============================================================
moc_para01,48,179,0	script	130-190 Mission Board	428,{

    if (BaseLevel < 130 || BaseLevel > 190) {
        mes "[130-190 Mission Board]";
        mes "This board contains elite contracts. Only adventurers between levels 130 and 190 may apply.";
        close;
    }

    // Check Cooldown (3 hours)
    if (eden_130_cooldown > checktime()) {
        .@delay = eden_130_cooldown - checktime();
        .@hours = .@delay / 3600;
        .@mins = (.@delay % 3600) / 60;
        mes "[130-190 Mission Board]";
        mes "You must rest before taking another high-risk contract.";
        mes "Time remaining: ^FF0000" + .@hours + "h " + .@mins + "m^000000.";
        close;
    }

    clearinit;
    mes "[130-190 Mission Board]";
    mes "Please select a mission category:";
    next;

    // Direct players based on their Base Level tier
    if (BaseLevel >= 130 && BaseLevel <= 159) {
        .@menu$ = "";
        if (checkquest(66001) == -1) .@menu$ = .@menu$ + "Accept Lv130+: Faceworm Hunt:";
        else if (checkquest(66001, HUNTING) == 2) .@menu$ = .@menu$ + "Complete Lv130+: Faceworm Hunt:";
        else .@menu$ = .@menu$ + "In Progress: Faceworm Hunt:";

        if (checkquest(66002) == -1) .@menu$ = .@menu$ + "Accept Lv130+: Lasagna Cleanup:";
        else if (checkquest(66002, HUNTING) == 2) .@menu$ = .@menu$ + "Complete Lv130+: Lasagna Cleanup:";
        else .@menu$ = .@menu$ + "In Progress: Lasagna Cleanup:";

        .@menu$ = .@menu$ + "Cancel";
        .@choice = select(.@menu$);

        switch(.@choice) {
            case 1:
                if (checkquest(66001) == -1) {
                    mes "[130-190 Mission Board]";
                    mes "Mission: Eliminate 30 Faceworms.";
                    next;
                    if (select("Accept:Leave") == 1) { setquest 66001; close; }
                } else if (checkquest(66001, HUNTING) == 2) {
                    mes "[130-190 Mission Board]";
                    mes "Contract completed successfully.";
                    completequest 66001;
                    getexp 35000000, 10000000; // Adjust custom reward values here
                    eden_130_cooldown = checktime() + 10800;
                    close;
                } else { mes "[130-190 Mission Board]"; mes "The hunt is still ongoing."; close; }
                break;

            case 2:
                if (checkquest(66002) == -1) {
                    mes "[130-190 Mission Board]";
                    mes "Mission: Eliminate 30 Sweet Lunatics.";
                    next;
                    if (select("Accept:Leave") == 1) { setquest 66002; close; }
                } else if (checkquest(66002, HUNTING) == 2) {
                    mes "[130-190 Mission Board]";
                    mes "Contract completed successfully.";
                    completequest 66002;
                    getexp 35000000, 10000000;
                    eden_130_cooldown = checktime() + 10800;
                    close;
                } else { mes "[130-190 Mission Board]"; mes "The hunt is still ongoing."; close; }
                break;
            case 3: close;
        }
    } else if (BaseLevel >= 160 && BaseLevel <= 190) {
        .@menu$ = "";
        if (checkquest(66003) == -1) .@menu$ = .@menu$ + "Accept Lv160+: Heart Hunter Hunt:";
        else if (checkquest(66003, HUNTING) == 2) .@menu$ = .@menu$ + "Complete Lv160+: Heart Hunter Hunt:";
        else .@menu$ = .@menu$ + "In Progress: Heart Hunter Hunt:";

        if (checkquest(66004) == -1) .@menu$ = .@menu$ + "Accept Lv160+: Scrap Robot Hunt:";
        else if (checkquest(66004, HUNTING) == 2) .@menu$ = .@menu$ + "Complete Lv160+: Scrap Robot Hunt:";
        else .@menu$ = .@menu$ + "In Progress: Scrap Robot Hunt:";

        .@menu$ = .@menu$ + "Cancel";
        .@choice = select(.@menu$);

        switch(.@choice) {
            case 1:
                if (checkquest(66003) == -1) {
                    mes "[130-190 Mission Board]";
                    mes "Mission: Eliminate 30 Heart Hunters in Werner Lab / Rudus.";
                    next;
                    if (select("Accept:Leave") == 1) { setquest 66003; close; }
                } else if (checkquest(66003, HUNTING) == 2) {
                    mes "[130-190 Mission Board]";
                    mes "Superb combat execution. Here is your reward.";
                    completequest 66003;
                    getexp 75000000, 30000000;
                    eden_130_cooldown = checktime() + 10800;
                    close;
                } else { mes "[130-190 Mission Board]"; mes "The targets are still active."; close; }
                break;

            case 2:
                if (checkquest(66004) == -1) {
                    mes "[130-190 Mission Board]";
                    mes "Mission: Eliminate 30 Scrap Robots in Einbech Dungeon 3.";
                    next;
                    if (select("Accept:Leave") == 1) { setquest 66004; close; }
                } else if (checkquest(66004, HUNTING) == 2) {
                    mes "[130-190 Mission Board]";
                    mes "Superb combat execution. Here is your reward.";
                    completequest 66004;
                    getexp 75000000, 30000000;
                    eden_130_cooldown = checktime() + 10800;
                    close;
                } else { mes "[130-190 Mission Board]"; mes "The targets are still active."; close; }
                break;
            case 3: close;
        }
    }
}
