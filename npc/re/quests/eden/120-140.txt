        mes "This mission is currently on an active 24-hour cooldown lock.";
        close;
    }
    if (checkquest(.@cd_id) == 2) {
        erasequest .@cd_id;
    }

    .@state = checkquest(.@q_id);
    if (.@state == 0 || .@state == 1) {
        mes "[Mission Board 120~140]";
        mes "You are already active on this hunting assignment.";
        close;
