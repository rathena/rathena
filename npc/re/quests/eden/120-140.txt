    .@mob_name$ = getarg(1);
    .@count = getarg(2);

        mes "[Mission Board 120~140]";
        mes "You are already hunting " + .@mob_name$ + ".";
        close;
    }
    if (.@state == 2) {
        mes "[Mission Board 120~140]";
        mes "You have already completed this task today.";
        close;
    }

    setquest .@q_id;
    mes "[Mission Board 120~140]";
    mes "Task Registered: Defeat " + .@count + " " + .@mob_name$ + ".";
    close;
