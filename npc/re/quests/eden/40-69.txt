//============================================================
// kRO Eden Group Leveling Board (Level 40 ~ 69)
// Simplified Version matching the working 1~39 structure
//============================================================

moc_para01,42,15,0	script	Mission Board 40~69	1121,{

    if (!callfunc("F_HasEdenGroupMark")) {
        mes "- You need to have an -";
        mes "- ^4d4dff'Eden Group Mark'^000000 -";
        mes "- to receive these missions. -";
        close;
    }

    if (BaseLevel < 40) {
        mes "[Mission Board 40~69]";
        mes "This board contains missions for intermediate adventurers.";
        mes "You must be at least Base Level 40.";
        close;
    }

    mes "[Mission Board 40~69]";
    mes "Please select a region to view available assignments:";
    next;
    
    switch(select("Shipwreck:Lighthouse:Orc Village:Turn in Completed Missions:Cancel")) {
        
        // ==========================================
        // SHIPWRECK
        // ==========================================
        case 1:
            mes "[Mission Board 40~69]";
            mes "Select a target to hunt in the Shipwreck:";
            next;
            switch(select("Hunt 20 Pirate Skeleton:Hunt 10 Kukre")) {
                case 1: callsub L_AssignQuest, 80030, "Pirate Skeleton", 20;
                case 2: callsub L_AssignQuest, 80031, "Kukre", 10;
            }
            break;

        // ==========================================
        // LIGHTHOUSE
        // ==========================================
        case 2:
            mes "[Mission Board 40~69]";
            mes "Select a target to hunt near the Lighthouse:";
            next;
            switch(select("Hunt 10 Aster:Hunt 10 Crab")) {
                case 1: callsub L_AssignQuest, 80032, "Aster", 10;
                case 2: callsub L_AssignQuest, 80033, "Crab", 10;
            }
            break;

        // ==========================================
        // ORC VILLAGE
        // ==========================================
        case 3:
            mes "[Mission Board 40~69]";
            mes "Select a target to hunt in the Orc Village:";
            next;
            switch(select("Hunt 20 Orc Skeleton:Hunt 20 Orc Zombie")) {
                case 1: callsub L_AssignQuest, 80034, "Orc Skeleton", 20;
                case 2: callsub L_AssignQuest, 80035, "Orc Zombie", 20;
            }
            break;

        // ==========================================
        // TURN IN COMPLETED MISSIONS
        // ==========================================
        case 4:
            mes "[Mission Board 40~69]";
            mes "Checking your quest progress...";
            next;
            
            // Loop through all quest IDs assigned to this board (80030 ~ 80035)
            for (.@i = 80030; .@i <= 80035; .@i++) {
                if (checkquest(.@i, HUNTING) == 2) {
                    callsub L_RewardQuest, .@i;
                    .@rewarded = 1;
                }
            }
            
            if (!.@rewarded) {
                mes "[Mission Board 40~69]";
                mes "You do not have any completed hunting missions to turn in yet.";
                close;
            }
            
            mes "[Mission Board 40~69]";
            mes "All completed tasks processed.";
            close;

        case 5:
            close;
    }
    end;

// ==========================================
// SUB-ROUTINES
// ==========================================
L_AssignQuest:
    .@q_id = getarg(0);
    .@mob_name$ = getarg(1);
    .@count = getarg(2);

    .@state = checkquest(.@q_id);
    if (.@state == 0 || .@state == 1) {
        mes "[Mission Board 40~69]";
        mes "You are already hunting " + .@mob_name$ + ".";
        close;
    }
    if (.@state == 2) {
        mes "[Mission Board 40~69]";
        mes "You have already completed this task today.";
        close;
    }

    setquest .@q_id;
    mes "[Mission Board 40~69]";
    mes "Task Registered: Defeat " + .@count + " " + .@mob_name$ + ".";
    close;

L_RewardQuest:
    .@q_id = getarg(0);
    
    // Base and Job Exp rewards multiplied by server rates
    switch(.@q_id) {
        case 80030: .@b_exp = 8600;  .@j_exp = 9600;  break; // Pirate Skeleton
        case 80031: .@b_exp = 6500;  .@j_exp = 6500;  break; // Kukre
        case 80032: .@b_exp = 8400;  .@j_exp = 8380;  break; // Aster
        case 80033: .@b_exp = 8400;  .@j_exp = 8380;  break; // Crab
        case 80034: .@b_exp = 14350; .@j_exp = 13700; break; // Orc Skeleton
        case 80035: .@b_exp = 14350; .@j_exp = 13700; break; // Orc Zombie
    }

    .@b_exp = (.@b_exp * getbattleflag("base_exp_rate")) / 100;
    .@j_exp = (.@j_exp * getbattleflag("job_exp_rate")) / 100;

    completequest .@q_id;
    getexp .@b_exp, .@j_exp;
    return;
}
