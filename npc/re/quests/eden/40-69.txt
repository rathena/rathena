L_AssignQuest:
    .@q_id = getarg(0); .@cd_id = getarg(1); .@mob_name\$ = getarg(2); .@min_lv = getarg(3);

    if (BaseLevel < .@min_lv) {
        mes "[Mission Board 40~69]";
        mes "You must be at least Base Level " + .@min_lv + " for this mission.";
        close;
    }
    if (checkquest(.@cd_id) == 0 || checkquest(.@cd_id) == 1) {
        mes "[Mission Board 40~69]";
        mes "This mission is currently on a 24-hour cooldown lock.";
        close;
    }
    if (checkquest(.@cd_id) == 2) erasequest .@cd_id;
    if (checkquest(.@q_id) == 0 || checkquest(.@q_id) == 1) {
        mes "[Mission Board 40~69]";
        mes "You are already active on this target assignment.";
        close;
    }

    setquest .@q_id;
    mes "[Mission Board 40~69]";
    mes "Task Confirmed: " + .@mob_name\$ + ". Good luck!";
    close;
