//============================================================
// kRO Eden Group Leveling Board (Level 1 ~ 39)
// Exact match with user image definitions & requirements
//============================================================

moc_para01,40,15,0	script	Mission Board 1~39	1121,{

    if (!callfunc("F_HasEdenGroupMark")) {
        mes "- You need to have an -";
        mes "- ^4d4dff'Eden Group Mark'^000000 -";
        mes "- to receive these missions. -";
        close;
    }

    if (BaseLevel < 1) {
        mes "[Mission Board 1~39]";
        mes "This board contains missions for novice adventurers.";
        close;
    }

    if (Eden_Board1_Cooldown > gettime(DT_DAYOFYEAR) && Eden_Board1_Year == gettime(DT_YEAR)) {
        mes "[Mission Board 1~39]";
        mes "You have already completed missions from this board today.";
        mes "Please come back tomorrow after 04:00 AM.";
        close;
    }

    mes "[Mission Board 1~39]";
    mes "Please select a region to view available hunting assignments:";
    next;
    
    switch(select("Payon Fields:Prontera Fields:Payon Dungeon:Morroc Fields:Comodo Dungeon:Turn in Completed Missions:Cancel")) {
        
        // ==========================================
        // PAYON FIELDS
        // ==========================================
        case 1:
            mes "[Mission Board 1~39]";
            mes "Select a target to hunt in the Payon fields:";
            next;
            switch(select("Hunt 5 Willows (Req: 3 Barren Trunk):Hunt 5 Boa:Hunt 5 Wormtail:Hunt 10 Spore")) {
                case 1: callsub L_AssignQuest, 80000, "Willow", 5, 15, 1; break;
                case 2: callsub L_AssignQuest, 800001, "Boa", 5, 15, 15; break;
                case 3: callsub L_AssignQuest, 800002, "Wormtail", 5, 15, 15; break;
                case 4: callsub L_AssignQuest, 800003, "Spore", 10, 15, 15; break;
            }
            break;

        // ==========================================
        // PRONTERA FIELDS
        // ==========================================
        case 2:
            mes "[Mission Board 1~39]";
            mes "Select a target to hunt in the Prontera fields:";
            next;
            switch(select("Hunt 10 Hornet:Hunt 10 Rocker:Hunt 10 Ambernite")) {
                case 1: callsub L_AssignQuest, 800004, "Hornet", 10, 15, 10; break;
                case 2: callsub L_AssignQuest, 800005, "Rocker", 10, 15, 10; break;
                case 3: callsub L_AssignQuest, 800006, "Ambernite", 10, 15, 10; break;
            }
            break;

        // ==========================================
        // PAYON DUNGEON
        // ==========================================
        case 3:
            mes "[Mission Board 1~39]";
            mes "Select a target to hunt in Payon Dungeon:";
            next;
            switch(select("Hunt 10 Zombie & 20 Skeleton")) {
                case 1: callsub L_AssignQuest, 800007, "Zombie & Skeleton", 10, 15, 15; break;
            }
            break;

        // ==========================================
        // MORROC FIELDS
        // ==========================================
        case 4:
            mes "[Mission Board 1~39]";
            mes "Select a target to hunt in the Morroc fields:";
            next;
            switch(select("Hunt 10 Muka & 20 Pecopeco")) {
                case 1: callsub L_AssignQuest, 800008, "Muka & Pecopeco", 10, 15, 20; break;
            }
            break;

        // ==========================================
        // COMODO DUNGEON
        // ==========================================
        case 5:
            mes "[Mission Board 1~39]";
            mes "Select a target to hunt in Comodo Dungeon:";
            next;
            switch(select("Hunt 10 Thara Frog (Req: 10 Sticky Webfoot):Hunt 10 Hydra (Req: 5 Tentacle):Hunt 10 Megalodon")) {
                case 1: callsub L_AssignQuest, 800009, "Thara Frog", 10, 15, 30; break;
                case 2: callsub L_AssignQuest, 800010, "Hydra", 10, 15, 30; break;
                case 3: callsub L_AssignQuest, 800011, "Megalodon", 10, 15, 30; break;
            }
            break;

        // ==========================================
        // TURN IN COMPLETED MISSIONS
        // ==========================================
        case 6:
            mes "[Mission Board 1~39]";
            mes "Checking your quest progress...";
            next;
            
            // Loop through all active quest IDs assigned to this board (80000 ~ 80011)
            for (.@i = 80000; .@i <= 80011; .@i++) {
                if (checkquest(.@i, HUNTING) == 2) {
                    
                    // Prior item check validation routine before rewarding
                    if (.@i == 80000 && countitem(1013) < 3) { // Barren Trunk ID: 1013
                        mes "^ff0000[Mission Failed]^000000";
                        mes "You finished hunting Willows, but you lack ^0000ff3 Barren Trunks^000000.";
                        next;
                        continue;
                    }
                    if (.@i == 80009 && countitem(948) < 10) { // Sticky Webfoot ID: 948
                        mes "^ff0000[Mission Failed]^000000";
                        mes "You finished hunting Thara Frogs, but you lack ^0000ff10 Sticky Webfoots^000000.";
                        next;
                        continue;
                    }
                    if (.@i == 80010 && countitem(953) < 5) { // Tentacle ID: 953
                        mes "^ff0000[Mission Failed]^000000";
                        mes "You finished hunting Hydras, but you lack ^0000ff5 Tentacles^000000.";
                        next;
                        continue;
                    }

                    // Delete item prior to finalizing completion
                    if (.@i == 80000) delitem 1013, 3;
                    if (.@i == 80009) delitem 948, 10;
                    if (.@i == 80010) delitem 953, 5;

                    callsub L_RewardQuest, .@i;
                    .@rewarded = 1;
                }
            }
            
            if (!.@rewarded) {
                mes "[Mission Board 1~39]";
                mes "You do not have any valid completed missions to turn in yet.";
                close;
            }
            
            mes "[Mission Board 1~39]";
            mes "All completed tasks processed successfully.";
            Eden_Board1_Cooldown = gettime(DT_DAYOFYEAR);
            Eden_Board1_Year = gettime(DT_YEAR);
            close;

        case 7:
            close;
    }
    end;

// ==========================================
// SUB-ROUTINES
// ==========================================
L_AssignQuest:
    .@q_id = getarg(0);
    .@mob_name$ = getarg(1);
    .@count = getarg(2);
    .@max_lv = getarg(3);
    .@min_lv = getarg(4);

    if (BaseLevel < .@min_lv) {
        mes "[Mission Board 1~39]";
        mes "You must be at least Base Level " + .@min_lv + " to accept this task.";
        close;
    }

    .@state = checkquest(.@q_id);
    if (.@state == 0 || .@state == 1) {
        mes "[Mission Board 1~39]";
        mes "You are already tracking " + .@mob_name$ + ".";
        close;
    }
    if (.@state == 2) {
        mes "[Mission Board 1~39]";
        mes "You have already completed this task today.";
        close;
    }

    setquest .@q_id;
    mes "[Mission Board 1~39]";
    mes "Task Registered: Defeat your targets for " + .@mob_name$ + ".";
    close;

L_RewardQuest:
    .@q_id = getarg(0);
    
    // Reward settings exactly aligned with the image values
    switch(.@q_id) {
        case 80000: .@b_exp = 800;  .@j_exp = 800;  break; // Willows
        case 80001: .@b_exp = 1600; .@j_exp = 1400; break; // Boa
        case 80002: .@b_exp = 1600; .@j_exp = 1400; break; // Wormtail
        case 80003: .@b_exp = 3000; .@j_exp = 2000; break; // Spore
        case 80004: .@b_exp = 1200; .@j_exp = 1000; break; // Hornet
        case 80005: .@b_exp = 1600; .@j_exp = 1400; break; // Rocker
        case 80006: .@b_exp = 2000; .@j_exp = 2000; break; // Ambernite
        case 80007: .@b_exp = 3200; .@j_exp = 3000; break; // Zombie & Skeleton
        case 80008: .@b_exp = 4800; .@j_exp = 4200; break; // Muka & Pecopeco
        case 80009: .@b_exp = 5500; .@j_exp = 5500; break; // Thara Frog
        case 80010: .@b_exp = 5500; .@j_exp = 5500; break; // Hydra
        case 80011: .@b_exp = 5500; .@j_exp = 5500; break; // Megalodon
    }

    .@b_exp = (.@b_exp * getbattleflag("base_exp_rate")) / 100;
    .@j_exp = (.@j_exp * getbattleflag("job_exp_rate")) / 100;

    completequest .@q_id;
    getexp .@b_exp, .@j_exp;
    return;
}
