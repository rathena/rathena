L_AssignQuest:
    .@q_id = getarg(0);
    .@cd_id = getarg(1);
    .@mob_name$ = getarg(2);
    .@count = getarg(3);

    if (checkquest(.@cd_id) == 0 || checkquest(.@cd_id) == 1) {
        mes "[Mission Board 70~99]";
        mes "This mission is currently on an active 24-hour cooldown lock.";
        close;
    }
    if (checkquest(.@cd_id) == 2) {
        erasequest .@cd_id;
    }

    .@state = checkquest(.@q_id);
    if (.@state == 0 || .@state == 1) {
        mes "[Mission Board 70~99]";
        mes "You are already tracking this specific assignment target.";
        close;
    }

    setquest .@q_id;
    mes "[Mission Board 70~99]";
    mes "Task Registered: Defeat " + .@count + " " + .@mob_name$ + ".";
    close;
