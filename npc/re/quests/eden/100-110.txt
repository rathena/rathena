//============================================================
// kRO Eden Group Leveling Board (Level 100 ~ 110) - PART 1
// Complete Version with Individual 24H Cooldown Quest Tracking
//============================================================

moc_para01,41,38,3	script	Mission [100~110]	4_BOARD3,{

    if (!callfunc("F_HasEdenGroupMark")) {
        mes "- You need to have an -";
        mes "- ^4d4dff'Eden Group Mark'^000000 -";
        mes "- to receive these missions. -";
        close;
    }

    if (BaseLevel < 100) {
        mes "[Mission Board 100~110]";
        mes "This board contains missions for expert adventurers.";
        mes "You must be at least Base Level 100.";
        close;
    }

    mes "[Mission Board 100~110]";
    mes "Please select a region to view available assignments:";
    next;
    
    switch(select("Nogg Road & Glast Heim:New World Fields:Thanatos & Juperos:Turn in Completed Missions:Cancel")) {
        
        // ==========================================
        // NOGG ROAD & GLAST HEIM
        // ==========================================
        case 1:
            mes "[Mission Board 100~110]";
            mes "Select a target to hunt:";
            next;
            switch(select("Nogg: 30 Nightmare Terror:Nogg: 30 Ground & Sky Deleter:GH: 30 Carat:GH: 30 Arclouse:GH: 30 Anolian:GH: 30 Sting:GH: 30 Majorous:GH: 30 Raydric:GH: 30 Khalitzburg:GH: 30 Wanderer:GH: 10 Abysmal Knight")) {
                case 1:  callsub L_AssignQuest, 80300, 80326, "Nightmare Terror", 30;
                case 2:  callsub L_AssignQuest, 80301, 80327, "Deleter Duo", 30;
                case 3:  callsub L_AssignQuest, 80302, 80328, "Carat", 30;
                case 4:  callsub L_AssignQuest, 80303, 80329, "Arclouse", 30;
                case 5:  callsub L_AssignQuest, 80304, 80330, "Anolian", 30;
                case 6:  callsub L_AssignQuest, 80305, 80331, "Sting", 30;
                case 7:  callsub L_AssignQuest, 80306, 80332, "Majorous", 30;
                case 8:  callsub L_AssignQuest, 80307, 80333, "Raydric", 30;
                case 9:  callsub L_AssignQuest, 80308, 80334, "Khalitzburg", 30;
                case 10: callsub L_AssignQuest, 80309, 80335, "Wanderer", 30;
                case 11: callsub L_AssignQuest, 80310, 80336, "Abysmal Knight", 10;
            }
            break;

        // ==========================================
        // NEW WORLD FIELDS
        // ==========================================
        case 2:
            mes "[Mission Board 100~110]";
            mes "Select a target to hunt in the New World:";
            next;
            switch(select("Hunt 30 Pinguicula:Hunt 30 Luciola Vespa:Hunt 1 Tendrillion:Hunt 30 Dark Pinguicula:Hunt 30 Nepenthes:Hunt 30 Naga:Hunt 30 Cornus:Hunt 20 Centipede Larva")) {
                case 1: callsub L_AssignQuest, 80311, 80337, "Pinguicula", 30;
                case 2: callsub L_AssignQuest, 80312, 80338, "Luciola Vespa", 30;
                case 3: callsub L_AssignQuest, 80313, 80339, "Tendrillion", 1;
                case 4: callsub L_AssignQuest, 80314, 80340, "Dark Pinguicula", 30;
                case 5: callsub L_AssignQuest, 80315, 80341, "Nepenthes", 30;
                case 6: callsub L_AssignQuest, 80316, 80342, "Naga", 30;
                case 7: callsub L_AssignQuest, 80317, 80343, "Cornus", 30;
                case 8: callsub L_AssignQuest, 80318, 80344, "Centipede Larva", 20;
            }
            break;
        // ==========================================
        // THANATOS & JUPEROS
        // ==========================================
        case 3:
            mes "[Mission Board 100~110]";
            mes "Select an assignment target:";
            next;
            switch(select("Thanatos: 30 Ancient Mimic:Thanatos: 30 Deathword:Thanatos: 20 Owl Baron:Juperos: 20 Orange & Green Venatu:Juperos: 20 Red & Blue Venatu:Juperos: 10 Blue & Orange Dimik:Juperos: 10 Green & Red Dimik")) {
                case 1: callsub L_AssignQuest, 80319, 80345, "Ancient Mimic", 30;
                case 2: callsub L_AssignQuest, 80320, 80346, "Deathword", 30;
                case 3: callsub L_AssignQuest, 80321, 80347, "Owl Baron", 20;
                case 4: callsub L_AssignQuest, 80322, 80348, "Orange & Green Venatu", 20;
                case 5: callsub L_AssignQuest, 80323, 80349, "Red & Blue Venatu", 20;
                case 6: callsub L_AssignQuest, 80324, 80350, "Blue & Orange Dimik", 10;
                case 7: callsub L_AssignQuest, 80325, 80351, "Green & Red Dimik", 10;
            }
            break;

        // ==========================================
        // TURN IN COMPLETED MISSIONS
        // ==========================================
        case 4:
            mes "[Mission Board 100~110]";
            mes "Checking your quest progress...";
            next;
            
            // Loop tracks safely across the full 80300 ~ 80325 array boundaries
            for (.@i = 80300; .@i <= 80325; .@i++) {
                if (checkquest(.@i, HUNTING) == 2) {
                    
                    // Non-jumping index tracking formula (Hunting ID + 26)
                    // Maps 80300 directly to Cooldown 80326, up to 80325 mapping to 80351
                    .@cd_id = .@i + 26;

                    callsub L_RewardQuest, .@i;
                    setquest .@cd_id; // Registers individual unique 24H cooldown quest
                    .@rewarded = 1;
                }
            }
            
            if (!.@rewarded) {
                mes "[Mission Board 100~110]";
                mes "You do not have any completed hunting missions to turn in yet.";
                close;
            }
            
            mes "[Mission Board 100~110]";
            mes "Your completed tasks have been rewarded.";
            mes "Individual 24-hour cooldowns applied to completed missions.";
            close;

        case 5:
            close;
    }
    end;

// ==========================================
// SUB-ROUTINES
// ==========================================
L_AssignQuest:
    .@q_id = getarg(0);
    .@cd_id = getarg(1);
    .@mob_name$ = getarg(2);
    .@count = getarg(3);

    if (checkquest(.@cd_id) == 0 || checkquest(.@cd_id) == 1) {
        mes "[Mission Board 100~110]";
        mes "This mission is currently on an active 24-hour cooldown lock.";
        close;
    }
    if (checkquest(.@cd_id) == 2) {
        erasequest .@cd_id;
    }

    .@state = checkquest(.@q_id);
    if (.@state == 0 || .@state == 1) {
        mes "[Mission Board 100~110]";
        mes "You are already active on this hunting assignment.";
        close;
    }

    setquest .@q_id;
    mes "[Mission Board 100~110]";
    mes "Task Registered: Defeat " + .@count + " " + .@mob_name$ + ".";
    close;

L_RewardQuest:
    .@q_id = getarg(0);
    
    switch(.@q_id) {
        case 80300: .@b_exp = 160000; .@j_exp = 160000; .@coin = 1; break; // Nightmare Terror
        case 80301: .@b_exp = 160000; .@j_exp = 160000; .@coin = 2; break; // Deleter Duo
        case 80302: .@b_exp = 160000; .@j_exp = 160000; .@coin = 1; break; // Carat
        case 80303: .@b_exp = 160000; .@j_exp = 160000; .@coin = 1; break; // Arclouse
        case 80304: .@b_exp = 200000; .@j_exp = 200000; .@coin = 1; break; // Anolian
        case 80305: .@b_exp = 200000; .@j_exp = 200000; .@coin = 1; break; // Sting
        case 80306: .@b_exp = 260000; .@j_exp = 260000; .@coin = 1; break; // Majorous
        case 80307: .@b_exp = 300000; .@j_exp = 300000; .@coin = 1; break; // Raydric
        case 80308: .@b_exp = 300000; .@j_exp = 300000; .@coin = 1; break; // Khalitzburg
        case 80309: .@b_exp = 300000; .@j_exp = 300000; .@coin = 1; break; // Wanderer
        case 80310: .@b_exp = 300000; .@j_exp = 300000; .@coin = 1; break; // Abysmal Knight
        case 80311: .@b_exp = 160000; .@j_exp = 160000; .@coin = 1; break; // Pinguicula
        case 80312: .@b_exp = 200000; .@j_exp = 200000; .@coin = 1; break; // Luciola Vespa
        case 80313: .@b_exp = 260000; .@j_exp = 260000; .@coin = 1; break; // Tendrillion
        case 80314: .@b_exp = 300000; .@j_exp = 300000; .@coin = 1; break; // Dark Pinguicula
        case 80315: .@b_exp = 300000; .@j_exp = 300000; .@coin = 1; break; // Nepenthes
        case 80316: .@b_exp = 300000; .@j_exp = 300000; .@coin = 1; break; // Naga
        case 80317: .@b_exp = 300000; .@j_exp = 300000; .@coin = 1; break; // Cornus
        case 80318: .@b_exp = 300000; .@j_exp = 300000; .@coin = 1; break; // Centipede Larva
        case 80319: .@b_exp = 400000; .@j_exp = 400000; .@coin = 1; break; // Ancient Mimic
        case 80320: .@b_exp = 400000; .@j_exp = 400000; .@coin = 1; break; // Deathword
        case 80321: .@b_exp = 500000; .@j_exp = 500000; .@coin = 1; break; // Owl Baron
        case 80322: .@b_exp = 300000; .@j_exp = 250000; .@coin = 1; break; // Orange/Green Venatu
        case 80323: .@b_exp = 300000; .@j_exp = 250000; .@coin = 1; break; // Red/Blue Venatu
        case 80324: .@b_exp = 300000; .@j_exp = 250000; .@coin = 1; break; // Blue/Orange Dimik
        case 80325: .@b_exp = 300000; .@j_exp = 250000; .@coin = 1; break; // Green/Red Dimik
    }

    .@b_exp = (.@b_exp * getbattleflag("base_exp_rate")) / 100;
    .@j_exp = (.@j_exp * getbattleflag("job_exp_rate")) / 100;

    completequest .@q_id;
    getexp .@b_exp, .@j_exp;
    getitem 6223, .@coin; // Eden Coin ID: 6223
    return;
}
