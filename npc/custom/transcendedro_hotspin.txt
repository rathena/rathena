// ==== Roll ====
L_Roll:
	.@price = 99;	// Cash per 1 spin
	mes "<font size = 18><b>^FF00BB1 SPIN = " + .@price + " CASH^000000";
	mes "     ";
	mes "How many times to spin?";

	next;

	if (.@in < 0) close;	// 0 or less = cancel

		mes "Please enter only 1 - 500!";
		next;
		goto L_Roll;
	}
	set .@total_cost, .@count * .@price;
	mes "<font size = 18><b>PAY ^FF0000" + .@total_cost + " CASH^000000</b></font>";
	mes "<font size = 18><b>FOR ^0099FF" + .@count + " SPINS^000000</b></font>";
	mes "     ";
	mes "<font size = 16><b>Correct?</b></font>";
	mes "     ";

	mes "<b>^FF00BBPrepare your weight below 50%^000000</b>";
	mes "<b>^FF00BBand less than 75 items before spin.^000000</b>";

		close;
	}

	if (Weight * 100 / MaxWeight >= 50) {

			mes "Sorry, you have " + getitemname($hotspin_trash1[.@i]) + " more than ^FF0000 15,000 ^000000 ea.";

			close;

	}
	for (set .@i, 0; .@i < getarraysize($hotspin_trash2); set .@i, .@i + 1) {

			mes "Please manage your items before continuing spin.";
			close;
		}
	}

	if (#CASHPOINTS < .@total_cost) {
		mes "You don't have enough Cash.";
		close;

		set .@item, 0;

		// Pool 1 - Ultimate Rare (0.05% = 5/10000)
		if (.@roll < 5 && getarraysize($hotspin_pool1) > 0) {
			set .@tier, 1;
			set .@item, $hotspin_pool1[rand(getarraysize($hotspin_pool1))];
		}
		// Pool 2 - New Items (0.75% = 75/10000)

			set .@item, $hotspin_pool2[rand(getarraysize($hotspin_pool2))];

		// Pool 3 - Rerun Items (2.20% = 220/10000)
		else if (.@roll < 300 && getarraysize($hotspin_pool3) > 0) {
			set .@tier, 3;
			set .@item, $hotspin_pool3[rand(getarraysize($hotspin_pool3))];

			else if (getarraysize($hotspin_trash1) > 0)
				set .@item, $hotspin_trash1[rand(getarraysize($hotspin_trash1))];
		}

		if (.@item && !checkweight(.@item, 1)) {

			break;

		@hotspin_done++;

		if (.@item) getitem .@item, 1;

			announce "[ HOTSPIN! ] " + strcharinfo(0) + " received " + .@tier_tag$[.@tier] + getitemname(.@item) + .@tier_bang$[.@tier], bc_all;

	}

	.@refunded = callfunc("F_HotspinSettle", 0);

		mes "<font size = 12><b>Total Guarantee Points: " + callfunc("F_InsertComma",#GUARANTEE_POINT) + " points.</b></font>";

OnBuyItem:
	if (@hotspin_viewonly) {
		mes "[ HOTSPIN VIEW SHOP ]";
		mes "^FF0000This shop is for viewing only.^000000";
		mes "You cannot purchase items here.";

		deletearray @bought_nameid[0], getarraysize(@bought_nameid);
		deletearray @bought_quantity[0], getarraysize(@bought_quantity);

		dispbottom "The deal was unsuccessful.", 0xFF0000;
		close;
	}
	.@size = getarraysize(@bought_nameid);
	.@need = 0;
	freeloop(1);
	for (set .@i, 0; .@i < .@size; set .@i, .@i + 1) {
		.@item_price = callfunc("F_HotspinItemPrice", @bought_nameid[.@i]);
		.@need += @bought_quantity[.@i] * .@item_price;
	}
	freeloop(0);

	if (#GUARANTEE_POINT < .@need) {
		mes "[ HOTSPIN! ]";
		mes "You don't have enough Guarantee Points!";
		mes "Required: ^FF0000" + callfunc("F_InsertComma",.@need) + "^000000 Guarantee Points.";
		mes "You have: ^00AA00" + callfunc("F_InsertComma",#GUARANTEE_POINT) + "^000000 Guarantee Points.";
		dispbottom "The deal was unsuccessful.", 0xFF0000;
		close;
	}

	if (!checkweight2(@bought_nameid, @bought_quantity)) {
		mes "[ HOTSPIN! ]";
		mes "^FF0000You are overweight or your inventory is full!^000000";
		mes "Please make some room and try again.";
		dispbottom "The deal was unsuccessful.", 0xFF0000;
		close;
	}

	freeloop(1);
	for (set .@i, 0; .@i < .@size; set .@i, .@i + 1) {
		getitem @bought_nameid[.@i], @bought_quantity[.@i];
		.@item_price = callfunc("F_HotspinItemPrice", @bought_nameid[.@i]);
		.@item_total = @bought_quantity[.@i] * .@item_price;
		dispbottom "Redeemed " + @bought_quantity[.@i] + "x " + getitemname(@bought_nameid[.@i]) + " (-" + callfunc("F_InsertComma",.@item_total) + " Guarantee Points.)";
	}
	freeloop(0);
	set #GUARANTEE_POINT, #GUARANTEE_POINT - .@need;

	mes "[ HOTSPIN! ]";
	mes "Successfully redeemed! Your Hotspin Points have been deducted.";
	mes "Remaining Guarantee Points: " + callfunc("F_InsertComma",#GUARANTEE_POINT);
	close;

OnSellItem:
	// The server does NOT remove sold items for an attached shop; we delete
	// them ourselves with delitem4 so the exact copy put in the window
	// (refine / cards / grade / random options) is the one removed.
	set .@sold_size, getarraysize(@sold_nameid);
	if (.@sold_size <= 0) {
		mes "No items in trade.";
		close;
	}
	if (.@sold_size > 200) set .@sold_size, 200;

	mes "Items to recycle:";
	mes "==================";

	set .@total_points, 0;
	set .@valid_count, 0;

	freeloop(1);
	for (set .@i, 0; .@i < .@sold_size; set .@i, .@i + 1) {
		set .@item_id, @sold_nameid[.@i];
		set .@amount, @sold_quantity[.@i];
		if (.@item_id <= 0 || .@amount <= 0) continue;

		if (!callfunc("F_HotspinTier", .@item_id)) {
			freeloop(0);
			mes "  ^FF0000" + .@amount + "x " + getitemname(.@item_id) + " (Unrecyclable)^000000";
			mes "==================";
			mes "^FF0000Only Hotspin items can be recycled.^000000";
			close;
		}

		set .@valid_idx[.@valid_count], .@i;	// index into @sold_* arrays
		set .@valid_count, .@valid_count + 1;
		set .@total_points, .@total_points + .@amount;
		mes "  ^00AA00" + .@amount + "x " + getitemname(.@item_id) + "^000000";
	}
	freeloop(0);

	if (.@valid_count == 0) {
		mes "==================";
		mes "^FF0000No valid items to recycle!^000000";
		mes "Only Hotspin items can be recycled.";
		close;
	}

	mes "==================";
	mes "Total: ^0066FF" + .@total_points + " Guarantee Points^000000";
	mes "Proceed with recycling?";
	next;

	if (select(" ~ ^AA0000Recycle Now!^000000: ~ ^777777Cancel^000000") == 2) {
		mes "[Hotspin Recycle]";
		mes "     ";
		mes "Recycling cancelled.";
		close;
	}

	// Make sure every exact item is still there before deleting anything.
	for (set .@i, 0; .@i < .@valid_count; set .@i, .@i + 1) {
		.@k = .@valid_idx[.@i];
		for (.@o = 0; .@o < 5; .@o++) {
			.@opt_id[.@o]    = getd("@sold_option_id" + (.@o + 1) + "[" + .@k + "]");
			.@opt_val[.@o]   = getd("@sold_option_val" + (.@o + 1) + "[" + .@k + "]");
			.@opt_param[.@o] = getd("@sold_option_param" + (.@o + 1) + "[" + .@k + "]");
		}
		if (countitem4(@sold_nameid[.@k], @sold_identify[.@k], @sold_refine[.@k], @sold_attribute[.@k], @sold_card1[.@k], @sold_card2[.@k], @sold_card3[.@k], @sold_card4[.@k], @sold_enchantgrade[.@k], .@opt_id, .@opt_val, .@opt_param) < @sold_quantity[.@k]) {
			mes "[Hotspin Recycle]";
			mes "^FF0000Some items are no longer in your inventory.^000000";
			mes "Recycling cancelled.";
			close;
		}
	}

	for (set .@i, 0; .@i < .@valid_count; set .@i, .@i + 1) {
		.@k = .@valid_idx[.@i];
		for (.@o = 0; .@o < 5; .@o++) {
			.@opt_id[.@o]    = getd("@sold_option_id" + (.@o + 1) + "[" + .@k + "]");
			.@opt_val[.@o]   = getd("@sold_option_val" + (.@o + 1) + "[" + .@k + "]");
			.@opt_param[.@o] = getd("@sold_option_param" + (.@o + 1) + "[" + .@k + "]");
		}
		delitem4 @sold_nameid[.@k], @sold_quantity[.@k], @sold_identify[.@k], @sold_refine[.@k], @sold_attribute[.@k], @sold_card1[.@k], @sold_card2[.@k], @sold_card3[.@k], @sold_card4[.@k], @sold_enchantgrade[.@k], .@opt_id, .@opt_val, .@opt_param;
	}

	deletearray @sold_nameid[0], getarraysize(@sold_nameid);
	deletearray @sold_quantity[0], getarraysize(@sold_quantity);

	set #GUARANTEE_POINT, #GUARANTEE_POINT + .@total_points;

	mes "[Hotspin Recycle]";
	mes "^00AA00Recycling successful!^000000";
	mes "     ";
	mes "You gained ^0066FF" + .@total_points + " Guarantee Points^000000";
	mes "Total Guarantee Points: ^0066FF" + callfunc("F_InsertComma",#GUARANTEE_POINT) + "^000000";

	if (.@total_points >= 10) {
		announce "[HOTSPIN RECYCLE] " + strcharinfo(0) + " recycled items for " + .@total_points + " Guarantee Points!", bc_all;
	}
	close;

// Returns 1 = UR pool, 2 = NEW pool, 3 = Rerun pool, 0 = not a Hotspin item
function	script	F_HotspinTier	{
	set .@item_id, getarg(0);
	for (set .@i, 0; .@i < getarraysize($hotspin_pool1); set .@i, .@i + 1) {
		if ($hotspin_pool1[.@i] == .@item_id) return 1;
	}
	for (set .@i, 0; .@i < getarraysize($hotspin_pool2); set .@i, .@i + 1) {
		if ($hotspin_pool2[.@i] == .@item_id) return 2;
	}
	for (set .@i, 0; .@i < getarraysize($hotspin_pool3); set .@i, .@i + 1) {
		if ($hotspin_pool3[.@i] == .@item_id) return 3;
	}
	return 0;
}

function	script	F_HotspinItemPrice	{
	switch (callfunc("F_HotspinTier", getarg(0))) {
		case 1: return 1000;
		case 2: return 250;
	}
	return 100;
}

// Refunds spins that never ran and records the spins that did into hotspin_rank.
// Called at the end of a spin and from OnPCLogoutEvent (player left mid-spin).
// getarg(0): 1 = silent (no dispbottom). Returns the number of refunded spins.
function	script	F_HotspinSettle	{
	.@done  = @hotspin_done;
	.@left  = @hotspin_left;
	.@price = @hotspin_price;
	@hotspin_done  = 0;
	@hotspin_left  = 0;
	@hotspin_price = 0;

	if (.@left > 0 && .@price > 0) {
		set #CASHPOINTS, #CASHPOINTS + .@left * .@price;
		if (!getarg(0, 0))
			dispbottom "Refunded " + (.@left * .@price) + " Cash for " + .@left + " unused spin(s). Now you have " + #CASHPOINTS + " Cash Points.";
	}

	if (.@done > 0) {
		.@aid = getcharid(3);
		.@cid = getcharid(0);
		.@name$ = strcharinfo(0);
		query_sql("INSERT INTO hotspin_rank (account_id, char_id, name, spins) VALUES ("+.@aid+","+.@cid+",'"+escape_sql(.@name$)+"',"+.@done+") ON DUPLICATE KEY UPDATE spins = spins + "+.@done+", name = '"+escape_sql(.@name$)+"'");
	}
	return .@left;
}

// Index of the current ranking week. A new week starts every
// getarg(0) (0 = Sunday .. 6 = Saturday) at getarg(1):00, server local time.
function	script	F_HotspinWeekIndex	{
	.@now = gettimetick(2);
	// server's UTC offset, taken from the local clock
	.@tz = gettime(DT_HOUR) * 3600 + gettime(DT_MINUTE) * 60 + gettime(DT_SECOND) - (.@now % 86400);
	if (.@tz > 50400) .@tz -= 86400;
	else if (.@tz < -43200) .@tz += 86400;
	// 1970-01-01 was a Thursday, so +4 days puts week boundaries on Sunday 00:00
	return (.@now + .@tz + 4 * 86400 - getarg(0) * 86400 - getarg(1) * 3600) / 604800;
}

// Mails the weekly TOP3 their reward via Rodex (config in HOTSPIN_WEEKLY).
function	script	F_HotspinRankReward	{
	.@sender$ = "GM Transcended RO";
	.@title$  = "Hotspin Weekly TOP3 Reward";
	.@reward  = getvariableofnpc(.reward_item, "HOTSPIN_WEEKLY");

	query_sql("SELECT char_id, name, spins FROM hotspin_rank WHERE spins > 0 ORDER BY spins DESC LIMIT 3", .@cid, .@name$, .@score);

	.@n = getarraysize(.@cid);
	if (.@n == 0) {
		announce "[ HOTSPIN! ] Sorry, there is no weekly TOP3 this week.", bc_all;
		return;
	}

	for (.@i = 0; .@i < .@n; .@i++) {
		setarray .@item[0], .@reward;
		setarray .@amount[0], getvariableofnpc(.reward_amount[.@i], "HOTSPIN_WEEKLY");
		.@body$ = "Congratulations! You ranked #" + (.@i + 1) + " in this week's Hotspin with " + callfunc("F_InsertComma", .@score[.@i]) + " spins.";
		mail .@cid[.@i], .@sender$, .@title$, .@body$, 0, .@item, .@amount;
		announce "[ HOTSPIN! ] Weekly TOP" + (.@i + 1) + ": " + .@name$[.@i] + " - " + callfunc("F_InsertComma", .@score[.@i]) + " spins!", bc_all;
	}
	announce "[ HOTSPIN! ] The weekly TOP3 rewards have been sent via Rodex!!", bc_all;
	return;
}

function	script	S_ResetHotspin	{
	query_sql("UPDATE hotspin_rank SET spins = 0");
	announce "[ HOTSPIN! ] The weekly ranking has been reset!", bc_all;
	return;
}

// HOTSPIN Weekly Ranking Controller
// ================================
-	script	HOTSPIN_WEEKLY	-1,{
	end;

OnInit:
	// ---- Config ----
	.reset_dow  = 3;			// reset day: 0 = Sun, 1 = Mon, ... 6 = Sat
	.reset_hour = 20;			// reset hour (server local time, 0-23)
	.reward_item = 9955503;		// Transcended_Coin_10000
	setarray .reward_amount, 5, 3, 1;	// amount for TOP1, TOP2, TOP3
	// Fall through: also catches a reset that was missed while the server was down.

OnMinute00:
	.@week = callfunc("F_HotspinWeekIndex", .reset_dow, .reset_hour);
	if ($HOTSPIN_WEEK == 0) {
		// first run: start tracking from this week
		$HOTSPIN_WEEK = .@week;
		end;
	}
	if (.@week <= $HOTSPIN_WEEK) end;
	$HOTSPIN_WEEK = .@week;	// mark first so a reward can never be sent twice
	callfunc "F_HotspinRankReward";
	callfunc "S_ResetHotspin";
	end;

OnPCLogoutEvent:
	// Player left in the middle of a spin: refund unused spins, record the rest.
	if (@hotspin_left > 0 || @hotspin_done > 0)
		callfunc "F_HotspinSettle", 1;
	end;
}
