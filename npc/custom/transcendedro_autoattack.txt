//===== TranscendedRO Script =================================
//= Auto Attack - settings menu
//===== Description: =========================================
//= @aa opens the settings window. The attack logic lives in
//= source: src/custom/autoattack.hpp
//= Script commands used: autoattack(), aa_skillinfo(), aa_applyloot()
//= Settings are permanent character variables AA_*, read live
//= by the source every tick.
//============================================================

-	script	autoattack_menu	-1,{

// Starts a fresh dialog page (clear needs an active mes)
function	Page	{
	mes "";
	clear;
	mes "[Auto Attack]";
	return;
}

function	OnOff	{
	if (getarg(0))
		return "[ON]";
	return "[OFF]";
}

function	IsVip	{
	return (vip_status(VIP_STATUS_ACTIVE) == 1 || getgroupid() == 99);
}

function	CanStart	{
	if (.require_vip && !IsVip()) {
		dispbottom "[Auto Attack] You must be a VIP to use Auto Attack.";
		return 0;
	}
	// Nested if: script && does not short-circuit, so countitem(0) would run and error.
	if (.ticket > 0) if (countitem(.ticket) <= 0) {
		dispbottom "[Auto Attack] You need a " + getitemname(.ticket) + ".";
		return 0;
	}
	return 1;
}

// select() splits options on ':' so names must not contain it
function	NoColon	{
	return replacestr(getarg(0), ":", " ");
}

function	ItemLabel	{
	if (getarg(0) <= 0)
		return "(empty)";
	return NoColon(getitemname(getarg(0))) + " [ID " + getarg(0) + "]";
}

function	SkillLabel	{
	if (getarg(0) <= 0)
		return "(empty)";
	.@lv$ = "Max";
	if (getarg(1) > 0)
		.@lv$ = "" + getarg(1);
	return NoColon(aa_skillinfo(getarg(0), 0)) + " Lv " + .@lv$;
}

// Buff slot timing text. getarg: skill id, slot index
function	BuffInfo	{
	.@id = getarg(0);
	if (.@id <= 0)
		return "";
	.@sec = AA_BUF_SEC[getarg(1)];
	.@mode = aa_skillinfo(.@id, 4);
	if (.@mode == 2)
		return " (keep spheres full)";
	if (.@mode == 4)
		.@t$ = " (toggle - keep ON)";
	else if (.@sec > 0)
		.@t$ = " (every " + .@sec + "s)";
	else if (.@mode == 3)
		.@t$ = " (every 10s)";
	else
		.@t$ = " (auto)";
	.@sc = aa_skillinfo(.@id, 3);
	if (.@sc > 0) if (getstatus(.@sc)) {
		.@left = getstatus(.@sc, 5);
		if (.@left > 0)
			.@t$ += " - " + (.@left / 1000) + "s left";
		else
			.@t$ += " - active";
	}
	return .@t$;
}

//==============================================================
OnMenu:
	if (!AA_INIT) {
		AA_INIT = 1;
		AA_NORMAL = 1;
		AA_TPDELAY = 3;
	}
	if (AA_LOOTMANAGED && !IsVip())
		aa_applyloot(0);

	while (1) {
		Page();
		.@running = autoattack(-1);
		if (.@running)
			mes "Status: ^00AA00Running^000000";
		else
			mes "Status: ^FF0000Stopped^000000";

		.@s = select(
			(.@running ? "Stop" : "Start") + " Auto Attack",
			"Normal Attack " + OnOff(AA_NORMAL),
			"Skill Attack",
			"Skill Buff",
			"Item Support",
			"Teleport",
			"Loot Items",
			"Close");

		switch (.@s) {
		case 1:
			if (.@running) {
				autoattack(0);
				break;
			}
			if (!CanStart())
				break;
			close2;
			autoattack(1);
			end;
		case 2:
			AA_NORMAL = !AA_NORMAL;
			break;
		case 3:
			callsub S_SkillSlots, "AA_ATK_SK", "AA_ATK_LV", 1;
			break;
		case 4:
			callsub S_SkillSlots, "AA_BUF_SK", "AA_BUF_LV", 2;
			break;
		case 5:
			callsub S_ItemMenu;
			break;
		case 6:
			callsub S_Teleport;
			break;
		case 7:
			callsub S_Loot;
			break;
		default:
			close;
		}
	}
	end;

//==============================================================
// Skill slots. getarg: skill var, level var, category (1 attack, 2 buff)
S_SkillSlots:
	.@sk$ = getarg(0);
	.@lv$ = getarg(1);
	.@cat = getarg(2);
	while (1) {
		Page();
		if (.@cat == 1) {
			mes "Skill Attack - used in slot order,";
			mes "skipping skills on cooldown / no SP.";
			if (!AA_NORMAL)
				mes "Normal attack is OFF: used only when no skill is set.";
		} else {
			mes "Skill Buff - checked in order 1 > 5.";
			mes "(auto) reads the real buff time and";
			mes "recasts just before it runs out.";
		}
		.@menu$ = "";
		for (.@i = 0; .@i < 5; .@i++) {
			.@id = getd(.@sk$ + "[" + .@i + "]");
			.@menu$ += "Slot " + (.@i + 1) + " - " + SkillLabel(.@id, getd(.@lv$ + "[" + .@i + "]"));
			if (.@cat == 2)
				.@menu$ += BuffInfo(.@id, .@i);
			.@menu$ += ":";
		}
		.@s = select(.@menu$ + "Back") - 1;
		if (.@s >= 5)
			return;
		callsub S_PickSkill, .@sk$, .@lv$, .@cat, .@s;
	}

// getarg: skill var, level var, category, slot index
S_PickSkill:
	.@sk$ = getarg(0);
	.@lv$ = getarg(1);
	.@cat = getarg(2);
	.@slot = getarg(3);

	getskilllist;
	.@n = 0;
	.@menu$ = "";
	for (.@i = 0; .@i < @skilllist_count; .@i++) {
		.@id = @skilllist_id[.@i];
		if (.@id <= 0 || @skilllist_lv[.@i] <= 0)
			continue;
		if (!(aa_skillinfo(.@id, 1) & .@cat))
			continue;
		.@ids[.@n] = .@id;
		.@lvs[.@n] = @skilllist_lv[.@i];
		.@n++;
		.@menu$ += NoColon(aa_skillinfo(.@id, 0)) + " (ID " + .@id + "):";
	}

	Page();
	if (!.@n) {
		mes "You have not learned any skill usable here.";
		next;
		return;
	}
	mes "Choose a skill for slot " + (.@slot + 1) + ".";
	.@s = select(.@menu$ + "Clear this slot:Back") - 1;
	if (.@s == .@n + 1)
		return;
	if (.@s == .@n) {
		setd .@sk$ + "[" + .@slot + "]", 0;
		setd .@lv$ + "[" + .@slot + "]", 0;
		if (.@cat == 2)
			AA_BUF_SEC[.@slot] = 0;
		return;
	}

	.@id = .@ids[.@s];
	.@max = .@lvs[.@s];
	.@menu$ = "";
	for (.@l = 1; .@l <= .@max; .@l++) {
		if (.@l == .@max)
			.@menu$ += "Level Max (" + .@l + ")";
		else
			.@menu$ += "Level " + .@l + ":";
	}
	Page();
	mes aa_skillinfo(.@id, 0) + " (SkillID " + .@id + ")";
	mes "Choose the skill level.";
	.@l = select(.@menu$);

	setd .@sk$ + "[" + .@slot + "]", .@id;
	setd .@lv$ + "[" + .@slot + "]", (.@l == .@max ? 0 : .@l); // 0 = always highest learned

	if (.@cat == 2) {
		AA_BUF_SEC[.@slot] = 0;
		.@mode = aa_skillinfo(.@id, 4);
		Page();
		mes aa_skillinfo(.@id, 0);
		if (.@mode == 2) {
			// Spheres / coins / charms / soul energy / elemental balls
			mes "Recast whenever you have fewer";
			mes "spheres than the maximum.";
			next;
		} else if (.@mode == 4) {
			mes "Toggle skill - turned ON when it is off,";
			mes "never toggled off by Auto Attack.";
			next;
		} else if (.@mode == 3) {
			// Heals / summons have no status to watch
			mes "This skill has no buff timer.";
			mes "Recast every how many seconds? (1 - 3600)";
			input .@sec;
			AA_BUF_SEC[.@slot] = cap_value(.@sec, 1, 3600);
		} else {
			mes "When should this buff be recast?";
			mes "Auto reads the buff's real remaining";
			mes "time (incl. duration bonuses).";
			if (select("Auto - read buff time (recommended):Every N seconds (manual)") == 2) {
				Page();
				mes "Recast every how many seconds? (1 - 3600)";
				input .@sec;
				AA_BUF_SEC[.@slot] = cap_value(.@sec, 1, 3600);
			}
		}
	}
	return;

//==============================================================
S_ItemMenu:
	while (1) {
		Page();
		mes "Item Support";
		switch (select("HP items:SP items:Other usable items:Back")) {
		case 1: callsub S_ItemSlots, "AA_HP_IT", "AA_HP_PCT", 1; break;
		case 2: callsub S_ItemSlots, "AA_SP_IT", "AA_SP_PCT", 2; break;
		case 3: callsub S_ItemSlots, "AA_OT_IT", "AA_OT_MS", 3; break;
		default: return;
		}
	}

// getarg: item var, value var, mode (1 HP %, 2 SP %, 3 interval ms)
S_ItemSlots:
	.@it$ = getarg(0);
	.@val$ = getarg(1);
	.@mode = getarg(2);
	while (1) {
		Page();
		if (.@mode == 1) mes "HP items - first slot you own is used.";
		if (.@mode == 2) mes "SP items - first slot you own is used.";
		if (.@mode == 3) mes "Other items - used every interval.";
		.@menu$ = "";
		for (.@i = 0; .@i < 5; .@i++) {
			.@item = getd(.@it$ + "[" + .@i + "]");
			.@v = getd(.@val$ + "[" + .@i + "]");
			.@menu$ += "Slot " + (.@i + 1) + " - " + ItemLabel(.@item);
			if (.@item > 0) {
				if (.@mode == 3)
					.@menu$ += " every " + .@v + " ms";
				else
					.@menu$ += " below " + .@v + "%";
			}
			.@menu$ += ":";
		}
		.@s = select(.@menu$ + "Back") - 1;
		if (.@s >= 5)
			return;

		if (.@mode == 3)
			callsub S_InputItem, 3, getd(.@it$ + "[" + .@s + "]");
		else
			callsub S_PickItem, .@mode;
		.@item = @aa_pick;
		if (.@item < 0)
			continue; // back
		if (.@item == 0) {
			setd .@it$ + "[" + .@s + "]", 0;
			setd .@val$ + "[" + .@s + "]", 0;
			continue;
		}

		Page();
		mes ItemLabel(.@item);
		if (.@mode == 3) {
			mes "Use every how many ms? (100 - 3600000)";
			input .@v;
			.@v = cap_value(.@v, 100, 3600000);
		} else {
			mes "Use when " + (.@mode == 1 ? "HP" : "SP") + " is below what %? (1 - 99)";
			input .@v;
			.@v = cap_value(.@v, 1, 99);
		}
		setd .@it$ + "[" + .@s + "]", .@item;
		setd .@val$ + "[" + .@s + "]", .@v;
	}

// Item picker from inventory. getarg: 1/2 healing items, 3 usable items, 4 any item.
// Result in @aa_pick: item id, 0 = clear slot, -1 = back
S_PickItem:
	.@mode = getarg(0);
	getinventorylist;
	.@n = 0;
	.@menu$ = "";
	for (.@i = 0; .@i < @inventorylist_count; .@i++) {
		.@id = @inventorylist_id[.@i];
		.@type = getiteminfo(.@id, ITEMINFO_TYPE);
		if (.@mode <= 2 && .@type != IT_HEALING)
			continue;
		if (.@mode == 3 && .@type != IT_HEALING && .@type != IT_USABLE && .@type != IT_DELAYCONSUME && .@type != IT_CASH)
			continue;
		if (.@n > 0 && inarray(.@ids[0], .@id) >= 0)
			continue;
		.@ids[.@n] = .@id;
		.@n++;
		.@menu$ += NoColon(getitemname(.@id)) + " [ID " + .@id + "]:";
	}

	Page();
	if (!.@n)
		mes "No matching item in your inventory.";
	else
		mes "Choose an item from your inventory.";
	.@s = select(.@menu$ + "Enter ItemID manually:Clear this slot:Back") - 1;

	if (.@s < .@n) {
		@aa_pick = .@ids[.@s];
		return;
	}
	if (.@s == .@n) {
		Page();
		mes "Enter ItemID:";
		input .@id;
		if (.@id <= 0 || getitemname(.@id) == "null") {
			mes "Unknown item.";
			next;
			@aa_pick = -1;
			return;
		}
		@aa_pick = .@id;
		return;
	}
	@aa_pick = (.@s == .@n + 1) ? 0 : -1;
	return;

// Typed ItemID entry. getarg: 3 usable items only, 4 any item; current item id.
// Result in @aa_pick: item id, 0 = clear slot, -1 = back / invalid
S_InputItem:
	.@mode = getarg(0);
	Page();
	if (getarg(1) > 0)
		mes "Current - " + ItemLabel(getarg(1));
	mes "Enter ItemID.";
	mes "0 = clear this slot, -1 = back.";
	input .@id, -1, 2147483647;
	if (.@id < 0) {
		@aa_pick = -1;
		return;
	}
	if (.@id == 0) {
		@aa_pick = 0;
		return;
	}
	if (getitemname(.@id) == "null") {
		Page();
		mes "Item ID " + .@id + " does not exist.";
		next;
		@aa_pick = -1;
		return;
	}
	if (.@mode == 3) {
		.@type = getiteminfo(.@id, ITEMINFO_TYPE);
		if (.@type != IT_HEALING && .@type != IT_USABLE && .@type != IT_DELAYCONSUME && .@type != IT_CASH) {
			Page();
			mes ItemLabel(.@id) + " is not a usable item.";
			next;
			@aa_pick = -1;
			return;
		}
	}
	@aa_pick = .@id;
	return;

//==============================================================
S_Teleport:
	while (1) {
		if (!AA_TPDELAY)
			AA_TPDELAY = 3;
		Page();
		mes "Teleport needs Fly Wing or the";
		mes "Teleport skill enabled below.";
		.@s = select(
			"Teleport delay - " + AA_TPDELAY + " Sec.",
			"Teleport by Item " + OnOff(AA_TPITEM),
			"Teleport by Skill " + OnOff(AA_TPSKILL),
			"Teleport away from Monsters (" + getarraysize(AA_AVOID) + ")",
			"Teleport away from Bosses " + OnOff(AA_AVOIDBOSS),
			"Back");
		switch (.@s) {
		case 1:
			setarray .@sec[0], 3, 5, 7, 10;
			AA_TPDELAY = .@sec[select("3 Sec.:5 Sec.:7 Sec.:10 Sec.") - 1];
			break;
		case 2:
			if (AA_TPITEM) {
				AA_TPITEM = 0;
			} else if (countitem(601) || rentalcountitem(12887)) {
				AA_TPITEM = 1;
			} else {
				Page();
				mes "You didn't have any Fly Wing in your Inventory.";
				next;
			}
			break;
		case 3:
			if (AA_TPSKILL) {
				AA_TPSKILL = 0;
			} else if ((eaclass() & EAJ_BASEMASK) == EAJ_ACOLYTE && getskilllv("AL_TELEPORT") > 0) {
				AA_TPSKILL = 1;
			} else {
				Page();
				mes "This option is only for Acolyte classes";
				mes "that have learned the Teleport skill.";
				next;
			}
			break;
		case 4:
			callsub S_Avoid;
			break;
		case 5:
			AA_AVOIDBOSS = !AA_AVOIDBOSS;
			break;
		default:
			return;
		}
	}

S_Avoid:
	while (1) {
		// Monsters currently on this map + those already in the list
		deletearray .@cls;
		.@n = 0;
		.@count = getmapunits(BL_MOB, strcharinfo(3), .@gids);
		for (.@i = 0; .@i < .@count && .@n < 60; .@i++) {
			getunitdata .@gids[.@i], .@d;
			.@c = .@d[UMOB_CLASS];
			if (.@c <= 0 || (.@n > 0 && inarray(.@cls[0], .@c) >= 0))
				continue;
			.@cls[.@n] = .@c;
			.@n++;
		}
		for (.@i = 0; .@i < getarraysize(AA_AVOID); .@i++) {
			if (.@n > 0 && inarray(.@cls[0], AA_AVOID[.@i]) >= 0)
				continue;
			.@cls[.@n] = AA_AVOID[.@i];
			.@n++;
		}

		.@menu$ = "";
		for (.@i = 0; .@i < .@n; .@i++) {
			.@on = (inarray(AA_AVOID[0], .@cls[.@i]) >= 0);
			.@menu$ += OnOff(.@on) + " " + NoColon(getmonsterinfo(.@cls[.@i], MOB_NAME)) + " (ID " + .@cls[.@i] + "):";
		}

		Page();
		mes "Teleport away when these monsters";
		mes "come into view. (max 30)";
		if (!.@n)
			mes "No monsters on this map.";
		.@s = select(.@menu$ + "Back") - 1;
		if (.@s >= .@n)
			return;

		.@c = .@cls[.@s];
		.@idx = inarray(AA_AVOID[0], .@c);
		if (.@idx >= 0) {
			deletearray AA_AVOID[.@idx], 1;
		} else if (getarraysize(AA_AVOID) >= 30) {
			Page();
			mes "The list is full (30).";
			next;
		} else {
			AA_AVOID[getarraysize(AA_AVOID)] = .@c;
		}
	}

//==============================================================
S_Loot:
	while (1) {
		.@vip = IsVip();
		if (AA_LOOTMANAGED && !.@vip)
			aa_applyloot(0);

		.@ids_on = 0;
		for (.@i = 0; .@i < 10; .@i++)
			if (AA_LOOTID[.@i] && AA_LOOTON[.@i])
				.@ids_on++;
		.@types_on = 0;
		for (.@i = 0; .@i < getarraysize(.loottype); .@i++)
			if (AA_LOOTTYPE & (1 << .loottype[.@i]))
				.@types_on++;

		Page();
		mes "Loot Items";
		if (!.@vip)
			mes "Autoloot options are for VIP only.";
		.@s = select(
			"Manual pick-up " + OnOff(AA_PICKUP),
			"Use 'GREED' pick-up " + OnOff(AA_GREED),
			(.@vip ? "[VIP] Autoloot all items " + OnOff(AA_LOOTALL) : ""),
			(.@vip ? "[VIP] Aloottype (" + .@types_on + " types ON)" : ""),
			(.@vip ? "[VIP] Alootid (" + .@ids_on + "/10 ON)" : ""),
			"Back");
		switch (.@s) {
		case 1:
			AA_PICKUP = !AA_PICKUP;
			break;
		case 2:
			if (AA_GREED) {
				AA_GREED = 0;
			} else if (getskilllv("BS_GREED") > 0) {
				AA_GREED = 1;
			} else {
				Page();
				mes "You don't have the Greed skill (BS_GREED).";
				next;
			}
			break;
		case 3:
			AA_LOOTALL = !AA_LOOTALL;
			AA_LOOTMANAGED = 1;
			aa_applyloot(1);
			break;
		case 4:
			callsub S_LootType;
			break;
		case 5:
			callsub S_LootId;
			break;
		default:
			return;
		}
	}

S_LootType:
	while (1) {
		Page();
		mes "Aloottype - loot these item types:";
		.@menu$ = "";
		for (.@i = 0; .@i < getarraysize(.loottype); .@i++)
			.@menu$ += OnOff(AA_LOOTTYPE & (1 << .loottype[.@i])) + " " + .lootname$[.@i] + ":";
		.@s = select(.@menu$ + "Back") - 1;
		if (.@s >= getarraysize(.loottype))
			return;
		AA_LOOTTYPE = AA_LOOTTYPE ^ (1 << .loottype[.@s]);
		AA_LOOTMANAGED = 1;
		aa_applyloot(1);
	}

S_LootId:
	while (1) {
		Page();
		mes "Alootid - loot these items (max 10):";
		.@menu$ = "";
		for (.@i = 0; .@i < 10; .@i++) {
			.@menu$ += (.@i + 1) + ". " + ItemLabel(AA_LOOTID[.@i]);
			if (AA_LOOTID[.@i])
				.@menu$ += " " + (AA_LOOTON[.@i] ? "ON" : "OFF");
			.@menu$ += ":";
		}
		.@s = select(.@menu$ + "Back") - 1;
		if (.@s >= 10)
			return;

		if (AA_LOOTID[.@s]) {
			Page();
			mes ItemLabel(AA_LOOTID[.@s]);
			.@back = 0;
			switch (select("Turn " + (AA_LOOTON[.@s] ? "OFF" : "ON") + ":Change item:Clear:Back")) {
			case 1:
				AA_LOOTON[.@s] = !AA_LOOTON[.@s];
				break;
			case 2:
				callsub S_InputItem, 4, AA_LOOTID[.@s];
				if (@aa_pick > 0) {
					AA_LOOTID[.@s] = @aa_pick;
					AA_LOOTON[.@s] = 1;
				} else if (@aa_pick == 0) {
					AA_LOOTID[.@s] = 0;
					AA_LOOTON[.@s] = 0;
				}
				break;
			case 3:
				AA_LOOTID[.@s] = 0;
				AA_LOOTON[.@s] = 0;
				break;
			default:
				.@back = 1;
				break;
			}
			if (.@back)
				continue;
		} else {
			callsub S_InputItem, 4, 0;
			if (@aa_pick <= 0)
				continue;
			AA_LOOTID[.@s] = @aa_pick;
			AA_LOOTON[.@s] = 1;
		}
		AA_LOOTMANAGED = 1;
		aa_applyloot(1);
	}

//==============================================================
OnPCLoginEvent:
	// VIP info arrives from the char-server shortly after login
	if (AA_LOOTMANAGED)
		addtimer 5000, strnpcinfo(3) + "::OnLoginLoot";
	end;

OnLoginLoot:
	aa_applyloot(IsVip());
	end;

OnInit:
	.require_vip = 0;   // 1 = only VIP can start auto attack
	.ticket = 0;        // item required to start (0 = none, e.g. 9950001)
