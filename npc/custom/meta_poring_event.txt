// +-----------------------------------------------------------------------------------------------
// |
// |	MetaPoringEvent
// |
// +-----------------------------------------------------------------------------------------------
// |	Description
// +-----------------------------------------------------------------------------------------------
// |	Starts an event in Geffen in which a large number of R_PORING and a single META_PORING
// |	are spawned at a given time of the week. mapflag mf_noskill is active during the event
// |	and all players in Geffen will receive @autoloot 100, including players who enter Geffen
// |	after the event already started. The event lasts 60 minutes or until the META_PORING is
// |	killed. Upon end of the event or the META_PORING is killed, a cleanup event is triggered
// |	which kills all monsters in Geffen, deactivates the mapflag mf_noskill and sets
// |	@autoloot 0 to all players on the server.
// +-----------------------------------------------------------------------------------------------
// |	Change History
// +-----------------------------------------------------------------------------------------------
// |	20261114	File Created
// |	20261114	Replaced addrid() with getunits() and for-loop
// |	20261114	General Bugfix, replaced getunits() with getmapunits()
// |	20261114	Change to array .@players
// |	20261114	Added Debug Output
// |	20261114	Removed Debug Output, changed Cleanup Routine to set @autoloot 0 to all
// |				players on the server, added 5 minute warning before end of the event
// +-----------------------------------------------------------------------------------------------

-	script	MetaPoringEvent	-1,{
	end;

// Allows a GM (Group 99) to trigger or stop the event anytime via Whisper
// Whisper "npc:MetaPoringEvent" with "start" or "stop"
OnWhisperGlobal:
	if (getgmlevel() < 99) end;

	if (@whispervar0$ == "start") {
		dispbottom "Manually starting the Meta Poring Event...";
		goto OnSat1800;
	}
	if (@whispervar0$ == "stop") {
		dispbottom "Manually stopping and cleaning up the Meta Poring Event...";
		goto OnTimer3600000;
	}
	dispbottom "Usage: Whisper 'start' to begin the event, or 'stop' to end it.";
	end;

OnSat1745:
    announce "The Poring Invasion in Geffen will begin in 15 minutes! Prepare yourselves!", bc_all, 0xFFFF00;
	end;

// Triggers every Saturday at 18:00 (6:00 PM server time)
OnSat1800:
	if (.event_active) end;
	.event_active = 1;

	// Enable No-Skill and LoadMapEvent mapflags in Geffen
	setmapflag "geffen", mf_noskill;
	setmapflag "geffen", mf_loadevent;

	// Announce event start
	announce "The Poring Invasion has begun in Geffen! Skills are disabled and Autoloot is active in Geffen!", bc_all, 0xFFFF00;
	announce "Find and defeat the Imposter Poring within 1 hour for an epic reward!", bc_all, 0xFFFF00;

	// Spawn 2000 R_PORING (#1725) and 1 META_PORING (#25000) in Geffen
	monster "geffen", 0, 0, "Poring", 1725, 2000, "MetaPoringEvent::OnFakePoringDead";
	monster "geffen", 0, 0, "Poring", 25000, 1, "MetaPoringEvent::OnMetaPoringDead";

	// Start the 1-hour (3,600,000 ms) countdown timer
	initnpctimer;

	// Give @autoloot 100 ONLY to players currently in Geffen (with Debug Output)
	.@count = getmapunits(BL_PC, "geffen", .@players);
	for (.@i = 0; .@i < .@count; .@i++) {
		if (attachrid(.@players[.@i])) {
			atcommand "@autoloot 100";
		}
	}
	detachrid;
	end;

OnFakePoringDead:
	end;

OnMetaPoringDead:
	if (!.event_active) end;

	.event_active = 0;
	stopnpctimer;

	announce strcharinfo(0) + " got a Midgard Coin!", bc_all, 0xFFFF00;

	callsub S_CleanupEvent;
	end;

OnTimer3300000:
	announce "The Poring Invasion in Geffen will end in 5 minutes! Hurry up!", bc_all, 0xFFFF00;
	end;

OnTimer3600000:
	stopnpctimer;

	if (!.event_active) end;

	.event_active = 0;
	announce "Time is up! The imposter escaped from Geffen!", bc_all, 0xFF5555;

	callsub S_CleanupEvent;
	end;

// Triggers whenever a player enters or leaves Geffen while mf_loadevent is active
OnPCLoadMapEvent:
	if (.event_active) {
		if (strcharinfo(3) == "geffen") {
			atcommand "@autoloot 100";
		} else {
			atcommand "@autoloot 0";
		}
	}
	end;

// Cleanup subroutine
S_CleanupEvent:
	killmonster "geffen", "MetaPoringEvent::OnFakePoringDead";
	killmonster "geffen", "MetaPoringEvent::OnMetaPoringDead";

	// Remove No-Skill and LoadMapEvent restrictions from Geffen
	removemapflag "geffen", mf_noskill;
	removemapflag "geffen", mf_loadevent;

	// Disable @autoloot for ALL online players on the server
	.@count = getunits(BL_PC, .@players);
	for (.@i = 0; .@i < .@count; .@i++) {
		if (attachrid(.@players[.@i])) {
			atcommand "@autoloot 0";
		}
	}
	detachrid;
	return;
}
