// rAthena Costume Recycler NPC (Item ID Version)
// ==========================================

prontera,155,199,4	script	Gacha Machine	2_VENDING_MACHINE1,{
    mes .npc_name$;
    mes "Hello, adventurer!";
    mes "Got bad luck with your Gacha Eggs?";
    mes "Don't let those junk costumes gather dust. I melt them down into precious Gacha Coins.";
    next;
    
    mes .npc_name$;
    mes "My current exchange rate is:";
    mes "^FF00001 Costume = " + .coins_reward + " " + getitemname(.coin_id) + "(s)^000000";
    mes "Would you like to recycle a costume?";
    next;

    if (select("Yes, please.:No, thank you.") == 2) {
        mes .npc_name$;
        mes "Suit yourself. Come back if your inventory gets too heavy!";
        close;
    }

    // Build selection menu dynamically using modern script-local variables
    .@menu_str$ = "";
    deletearray .@mapped_index[0], getarraysize(.@mapped_index);
    .@menu_count = 0;

    for (.@i = 0; .@i < getarraysize(.valid_costumes); .@i++) {
        .@item_id = .valid_costumes[.@i];
        
        // FIXED: Subtract the equipped status (1 if equipped, 0 if not) from total count.
        // This isolates only the duplicate copies sitting in the inventory.
        .@qty = countitem(.@item_id) - isequipped(.@item_id);

        if (.@qty > 0) {
            // Shortened layout to "[Qty] Item Name" to prevent client UI text splitting
            .@menu_str$ += "[" + .@qty + "] " + getitemname(.@item_id) + ":";
            .@mapped_index[.@menu_count] = .@i;
            .@menu_count++;
        }
    }

    // No unequipped valid costumes found in inventory
    if (.@menu_count == 0) {
        mes .npc_name$;
        mes "It looks like you don't have any of the accepted costumes in your inventory right now.";
        mes "^FF0000Note: If you are wearing the costume, you must unequip it first!^000000";
        close;
    }

    // Append Cancel option tightly
    .@menu_str$ += "Cancel";

    mes .npc_name$;
    mes "Please select the costume you want to scrap:";
    next;

    .@choice = select(.@menu_str$) - 1;

    // Player clicked 'Cancel'
    if (.@choice == .@menu_count) {
        mes .npc_name$;
        mes "Change your mind? Alright then.";
        close;
    }

    // Direct lookup of the selected item ID from the mapped index array
    .@target_id = .valid_costumes[.@mapped_index[.@choice]];

    mes .npc_name$;
    mes "You selected: ^990000" + getitemname(.@target_id) + "^000000.";
    mes "Are you absolutely sure? This action ^FF0000cannot be undone^000000.";
    next;

    if (select("Yes, melt it down!:Wait, cancel!") == 2) {
        mes .npc_name$;
        mes "Close call! Your costume is safe.";
        close;
    }

    // Exploit/Lag double check right before deletion
    // Ensures they have at least 1 loose copy available to scrap
    if ((countitem(.@target_id) - isequipped(.@target_id)) < 1) {
        mes .npc_name$;
        mes "Hey! You don't have an unequipped copy of that costume anymore.";
        close;
    }

    // Execution
    delitem .@target_id, 1;
    getitem .coin_id, .coins_reward;
    
    // Aesthetic effects
    specialeffect2 EF_REFINE_SUCCESS, AREA, player;
    soundeffect "blacksmith.wav", 0;

    mes .npc_name$;
    mes "Done! Your costume has been recycled.";
    mes "Here is your ^009900" + .coins_reward + " " + getitemname(.coin_id) + "^000000.";
    close;

OnInit:
    .npc_name$ = "^0000FF[Costume Recycler]^000000";
    .coins_reward = 1; // Number of coins given per costume
    .coin_id = 50003;  // Your Gacha Coin ID

    // Add the numerical Item IDs of all valid costumes here
    setarray .valid_costumes[0], 
        31436, // Replace these numbers with your actual Costume Item IDs
        31771, 
        19551, 
        19888, 
        19886,
	19889,
	410138,
	480461,
	20127,
	20143,
	20151,
	19894,
	20280,
	31683,
	20130,
	20132,
	410082,
	20577,
	20592,
        19891;
    end;
}
