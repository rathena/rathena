prt_in,243,168,4	script	Stylist#prontera	4_F_02,{
	.@outfit = callsub(S_GetOutfit);
	.@body = getlook(LOOK_BODY2);
	.@second = (.@outfit && .@body == .@outfit);

	if (!.@outfit) {
		mes "[Jovovich]";
		mes "Come here a second.";
		mes "Turn. Let me look at you.";
		next;
		mes "[Jovovich]";
		mes "Needle, thread, a second set of clothes...";
		mes "I had a whole speech ready.";
		next;
		mes "[Jovovich]";
		mes "Those clothes already look right on you.";
		mes "I'm not touching the cut.";
		mes "If you want a different color, Ginedin beside me still dyes.";
		close;
	}

	mes "[Jovovich]";
	mes "Come here a second.";
	mes "Turn. Let me look at you.";
	next;
	mes "[Jovovich]";
	mes "I can change those clothes for you.";
	mes "The alternative set, or back to the usual ones.";
	mes "Interested?";
	next;
	if (select("I'm interested:Not now") == 2) {
		mes "[Jovovich]";
		mes "Come back when those clothes itch.";
		mes "Color is Ginedin's counter, not mine.";
		close;
	}

	mes "[Jovovich]";
	mes "One " + getitemname(.TicketId) + " for the switch.";
	mes "Do you have one on you?";
	next;
	if (select("I have it:I don't") == 2) {
		mes "[Jovovich]";
		mes "Then come back with a " + getitemname(.TicketId) + ".";
		mes "Rooke in the black market sells them.";
		close;
	}
	if (countitem(.TicketId) < 1) {
		mes "[Jovovich]";
		mes "You don't. Don't bluff a stylist.";
		mes "Rooke sells those tickets.";
		close;
	}

	delitem .TicketId, 1;
	if (.@second)
		setlook LOOK_BODY2, Class;
	else
		setlook LOOK_BODY2, .@outfit;

	mes "[Jovovich]";
	if (.@second)
		mes "There. The usual clothes, same as before.";
	else
		mes "There. The alternative cut.";
	mes "Bring another ticket when you want to switch again.";
	mes "A new color is Ginedin, next to me.";
	close;

S_GetOutfit:
	.@i = inarray(.Jobs[0], Class);
	if (.@i < 0)
		return 0;
	return .Outfits[.@i];

	.TicketId = 6959;
	setunittitle getnpcid(0), "<Second Costume>";
	setarray .Jobs[0],
		Job_Rune_Knight, Job_Rune_Knight2, Job_Rune_Knight_T, Job_Rune_Knight_T2, Job_Baby_Rune_Knight, Job_Baby_Rune_Knight2,

		Job_Guillotine_Cross, Job_Guillotine_Cross_T, Job_Baby_Guillotine_Cross,
		Job_Royal_Guard, Job_Royal_Guard2, Job_Royal_Guard_T, Job_Royal_Guard_T2, Job_Baby_Royal_Guard, Job_Baby_Royal_Guard2,
		Job_Sorcerer, Job_Sorcerer_T, Job_Baby_Sorcerer,
