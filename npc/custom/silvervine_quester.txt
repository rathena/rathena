prontera,153,191,4	script	Silvervine Quester	4_M_MERCAT1,{

    // ==========================================
    // DIALOGUE CORE LOGIC
    // ==========================================
    mes "[Silvervine Quester]";
    mes "Greetings! I hand out rewards to players who can take down elusive mini-bosses.";
    next;

    // Daily Limit Check & Reset
    if (#Silvervine_LastDate != gettime(DT_YYYYMMDD)) {
        #Silvervine_LastDate = gettime(DT_YYYYMMDD);
        #Silvervine_DailyCount = 0;
    }

    // Evaluate current state across native arrays
    .@active_index = -1;
    .@complete_index = -1;

    for (.@i = 0; .@i < .quest_count; .@i = .@i + 1) {
        .@q_id = .quests[.@i * 2];
        .@status = checkquest(.@q_id, HAVEQUEST); 

        if (.@status == 2 || .@status == 3) { // Catching both completion states safely
            .@complete_index = .@i;
            break;
        }
        if (.@status == 1) { // 1 = Quest is Active but objectives aren't finished yet
            .@active_index = .@i;
            break;
        }
    }

    // Hand in complete quest rewards
    if (.@complete_index != -1) {
        .@q_id = .quests[.@complete_index * 2];

        mes "[Silvervine Quester]";
        mes "Superb work! You successfully hunted down the target.";
        erasequest .@q_id;
        
        getitem .reward_item, .reward_amount;
        #Silvervine_DailyCount = #Silvervine_DailyCount + 1;
        
        mes "Received: ^00FF00" + .reward_amount + "x Silvervine Fruit^000000";
        mes "Completed Today: ^00FF00" + #Silvervine_DailyCount + "/" + .max_daily_quests + "^000000";
        close;
    }

    // Block if user is already holding an incomplete target assignment
    if (.@active_index != -1) {
        mes "[Silvervine Quester]";
        mes "Your current target is: ^FF0000" + .quests[(.@active_index * 2) + 1] + "^000000.";
        mes "Please eliminate them and check back with me.";
        close;
    }

    // Enforce daily hard ceiling
    if (#Silvervine_DailyCount >= .max_daily_quests) {
        mes "[Silvervine Quester]";
        mes "You have reached your limit of ^FF0000" + .max_daily_quests + " hunts^000000 for today.";
        mes "Come back tomorrow after midnight server time!";
        close;
    }

    // Roll a new contract
    mes "[Silvervine Quester]";
    mes "Are you ready to hunt a random mini-boss?";
    mes "Progress today: " + #Silvervine_DailyCount + " / " + .max_daily_quests;
    next;

    if (select("Give me a target!:Maybe later.") == 2) {
        mes "[Silvervine Quester]";
        mes "Be safe out there.";
        close;
    }

    // Secure native random generation mapping
    .@rand = rand(.quest_count);
    .@chosen_quest = .quests[.@rand * 2];
    .@monster_name = .quests[(.@rand * 2) + 1];

    setquest .@chosen_quest;

    mes "[Silvervine Quester]";
    mes "Your new assignment: ^FF0000" + .@monster_name + "^000000.";
    mes "Track your kill status directly inside your ^0000FFQuest Log (Alt+U)^000000.";
    close;

OnInit:
    // ==========================================
    // CONFIGURATION
    // ==========================================
    .max_daily_quests = 10;
    .reward_item = 6707; // Silvervine Fruit ID
    .reward_amount = 5;

    // Fixed the broken gap/newlines in the array setup to ensure solid index mapping
    setarray .quests[0], 
        99001, "Angeling",
        99002, "Archangelig",
        99003, "Cat O' Nine Tails",
        99004, "Deviling",
        99005, "Dragon Fly",
        99006, "Eclipse",
        99007, "Ghostring",
        99008, "Mastering",
        99009, "Maya Purple",
        99010, "Toad",
        99011, "Vagabond Wolf",
        99012, "Vocal";

    .quest_count = getarraysize(.quests) / 2;
    end;
}
