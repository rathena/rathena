//===== OzRo Script ==========================================
//= Garrett el Recolector
//===== Description: =========================================
//= Compra chatarra diaria a buen precio. Karma por PJ.
//= Moneda bronce cada 10 tratos unicos (1 por item del dia, max 2/dia).
//============================================================

geffen,143,70,3	script	Garrett	1_M_SIGN1,{

	.@weekday = gettime(DT_DAYOFWEEK);

	if (.@weekday != TRSH_LastDay || TRSH_Item0$ == "" || TRSH_Item1$ == "") {
		TRSH_LastDay = .@weekday;
		TRSH_DailySpend = 0;
		TRSH_DoneItem0 = 0;
		TRSH_DoneItem1 = 0;
		.@size0 = getarraysize(.common_items$);
		.@size1 = getarraysize(.misc_items$);
		TRSH_Item0$ = .common_items$[rand(.@size0)];
		do {
			TRSH_Item1$ = .misc_items$[rand(.@size1)];
		} while (TRSH_Item1$ == TRSH_Item0$);
	}

	if (TRSH_DailySpend >= .daily_spending_limit) {
		mes .npc_name$;
		mes "Amigo, ya no me queda una moneda en la bolsa hoy.";
		mes "Vuelve manana. Siempre hay basura nueva que vale la pena.";
		close;
	}

	.@price0 = getiteminfo(TRSH_Item0$, ITEMINFO_SELL) * .increment;
	if (.@price0 < .min_price) .@price0 = .min_price;
	if (.@price0 > .max_price) .@price0 = .max_price;

	.@price1 = getiteminfo(TRSH_Item1$, ITEMINFO_SELL) * .increment;
	if (.@price1 < .min_price) .@price1 = .min_price;
	if (.@price1 > .max_price) .@price1 = .max_price;

	mes .npc_name$;
	if (OZRO_KARMA < 0) {
		mes "Tu otra vez...";
		mes "No me gusta tu aura, pero el negocio es el negocio.";
	} else if (OZRO_KARMA > 0) {
		mes "Ah, " + strcharinfo(0) + "! Me alegra verte.";
		mes "Tienes buen ojo para lo que vale la pena, eso se nota.";
	} else {
		mes "Bienvenido, forastero.";
		mes "Dicen que la basura de un hombre es el tesoro de otro.";
	}
	mes "Hoy tengo el ojo puesto en cosas interesantes.";
	next;

	mes .npc_name$;
	mes "Me quedan " + (.daily_spending_limit - TRSH_DailySpend) + " zeny para comprar hoy.";
	next;

	mes .npc_name$;
	mes "Busco:";
	mes "- " + getitemname(TRSH_Item0$) + " - " + .@price0 + " zeny c/u";
	mes "- " + getitemname(TRSH_Item1$) + " - " + .@price1 + " zeny c/u";
	next;

	switch (select("Comerciar", "Sobre tu negocio", "Marcharme")) {
	case 1:
		.@menu$ = getitemname(TRSH_Item0$) + ":" + getitemname(TRSH_Item1$) + ":Cambie de opinion";
		.@selected = select(.@menu$) - 1;
		if (.@selected == 2) {
			mes .npc_name$;
			mes "Sin prisa. Los buenos tratos requieren calma.";
			close;
		}

		if (.@selected == 0) {
			.@item$ = TRSH_Item0$;
			.@price = .@price0;
		} else {
			.@item$ = TRSH_Item1$;
			.@price = .@price1;
		}

		mes .npc_name$;
		mes "Cuantos " + getitemname(.@item$) + " me vendes?";
		mes "Tienes " + countitem(.@item$) + " en el inventario.";
		next;

		input .@amount;
		if (.@amount <= 0) {
			mes .npc_name$;
			mes "Vamos, hablemos en serio.";
			close;
		}
		if (countitem(.@item$) < .@amount) {
			mes .npc_name$;
			mes "No tienes tantos. Solo llevas " + countitem(.@item$) + ".";
			next;
			if (select("Vendo los que tengo", "Mejor despues") == 2) {
				mes .npc_name$;
				mes "Vuelve cuando tengas mas.";
				close;
			}
			.@amount = countitem(.@item$);
			if (.@amount <= 0) close;
		}

		.@total = .@price * .@amount;
		if (TRSH_DailySpend + .@total > .daily_spending_limit) {
			mes .npc_name$;
			mes "Eso pasa mi presupuesto de hoy.";
			mes "Solo puedo gastar " + (.daily_spending_limit - TRSH_DailySpend) + " zeny mas.";
			close;
		}

		mes .npc_name$;
		mes .@amount + " x " + getitemname(.@item$) + " = " + .@total + " zeny. Trato?";
		next;
		if (select("Trato hecho", "Paso") == 2) {
			mes .npc_name$;
			mes "La puerta sigue abierta.";
			close;
		}

		delitem .@item$, .@amount;
		Zeny += .@total;
		TRSH_DailySpend += .@total;
		TRSH_TotalSpent += .@total;
		TRSH_TRADES++;

		.@is_item0 = (.@item$ == TRSH_Item0$);
		.@is_item1 = (.@item$ == TRSH_Item1$);
		.@first_trade_today = 0;
		if (.@is_item0 && !TRSH_DoneItem0) {
			TRSH_DoneItem0 = 1;
			.@first_trade_today = 1;
		} else if (.@is_item1 && !TRSH_DoneItem1) {
			TRSH_DoneItem1 = 1;
			.@first_trade_today = 1;
		}

		if (.@first_trade_today) {
			TRSH_UniqueTrades++;
			OZRO_KARMA += .karma_per_trade;
		}

		mes .npc_name$;
		mes "Un placer hacer negocios.";
		if (TRSH_DailySpend >= .daily_spending_limit) {
			mes "Con eso agoto el presupuesto de hoy.";
			mes "Manana habra mas oportunidades.";
		}

		if (.@first_trade_today && TRSH_UniqueTrades % .trades_for_coin == 0) {
			getitem "Ozro_Bronze_Coin", 1;
			mes "Por cierto... eres un sujeto agradable.";
			mes "Creo que esto te sirve mas a ti que a mi.";
		}
		break;

	case 2:
		mes .npc_name$;
		mes "Llevo " + TRSH_TotalSpent + " zeny invertidos en chatarra de aventureros.";
		mes "Y contigo he cerrado " + TRSH_TRADES + " tratos. Cada uno cuenta.";
		break;

	case 3:
		mes .npc_name$;
		mes "Buen viaje. La basura de hoy puede ser la riqueza de manana.";
		break;
	}
	close;

OnInit:
	.npc_name$ = "[ ^0000FFGarrett el Recolector^000000 ]";
	.increment = 10;
	.min_price = 500;
	.max_price = 2000;
	.daily_spending_limit = 300000;
	.trades_for_coin = 10;
	.karma_per_trade = 1;

	setarray .common_items$[0],
		"Jellopy",          // 909

		"Scell",            // 911
