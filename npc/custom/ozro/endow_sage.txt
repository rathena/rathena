//===== OzRo Script ==========================================
//= Sabio de los Elementos
//===== Description: =========================================
//= Imbuye propiedad elemental al arma (endow) a cambio de
//= zeny y piedra elemental.
//============================================================

izlude,131,142,4	script	EndowedSage	2_M_SAGE_OLD,{

	mes .npc_name$;
	mes "Ah, un buscador de poder se acerca.";
	mes "Deseas canalizar las fuerzas de la naturaleza en tu arma?";
	mes "Solo te costara " + .zeny_cost + " zeny y la piedra elemental correspondiente.";
	next;

	if (Zeny < .zeny_cost) {
		mes .npc_name$;
		mes "Tsk, tsk. El poder no es para los que no tienen monedas.";
		mes "Vuelve cuando puedas pagar las bendiciones de la naturaleza.";
		close;
	}

	mes .npc_name$;
	mes "Dime, que elemento buscas?";
	next;

	switch (select("Fuego ardiente", "Viento cortante", "Hielo gelido", "Tierra firme")) {
	case 1:
		.@item$ = "Boody_Red"; // Red Blood
		.@sc_id = SC_FIREWEAPON;
		.@skill_id = SA_FLAMELAUNCHER;
		mes .npc_name$;
		mes "El fuego, esencia del caos y la destruccion. Buena eleccion.";
		break;
	case 2:
		.@item$ = "Wind_Of_Verdure"; // Wind of Verdure
		.@sc_id = SC_WINDWEAPON;
		.@skill_id = SA_LIGHTNINGLOADER;
		mes .npc_name$;
		mes "El viento, fuerza invisible que dobla la voluntad. Excelente.";
		break;
	case 3:
		.@item$ = "Crystal_Blue"; // Crystal Blue
		.@sc_id = SC_WATERWEAPON;
		.@skill_id = SA_FROSTWEAPON;
		mes .npc_name$;
		mes "El agua, molde silencioso del mundo. Sabia decision.";
		break;
	case 4:
		.@item$ = "Yellow_Live"; // Green Live
		.@sc_id = SC_EARTHWEAPON;
		.@skill_id = SA_SEISMICWEAPON;
		mes .npc_name$;
		mes "La tierra, cimiento de toda vida. Noble eleccion.";
		break;
	}
	next;

	if (countitem(.@item$) < 1) {
		mes .npc_name$;
		mes "Pides el poder de la naturaleza sin la ofrenda adecuada?";
		mes "Vete, y vuelve con la piedra que corresponde.";
		close;
	}

	delitem .@item$, 1;
	Zeny -= .zeny_cost;
	sc_start .@sc_id, 1800000, 1;
	skilleffect .@skill_id, 1;
	mes .npc_name$;
	mes "Listo. Que tus enemigos tiemblen ante tu arma.";
	mes "Ve y trae equilibrio... o caos, si eso prefieres.";
	close;

OnInit:
	.npc_name$ = "[ ^0000FFSabio de los Elementos^000000 ]";
	.zeny_cost = 15000;
	end;
}

prontera,159,186,4	duplicate(EndowedSage)	EndowedSage#protera	2_M_SAGE_OLD
morocc,168,106,4	duplicate(EndowedSage)	EndowedSage#morocc	2_M_SAGE_OLD
geffen,125,68,4	duplicate(EndowedSage)	EndowedSage#geffen	2_M_SAGE_OLD
