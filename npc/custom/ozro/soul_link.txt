//===== OzRo Script ==========================================
//= Kessinger, Vinculador de Almas
//===== Description: =========================================
//= Soul Link de pago. Espiritu segun BaseJob del personaje.
//= Futuro: quest alarga .duration en OnInit (ver 04-npcs.md).
//============================================================

yuno,145,185,4	script	Kessinger	4_M_ROTERT,{

	.@name$ = "[ ^0000FFKessinger, Vinculador de Almas^000000 ]";

	mes .@name$;
	mes "Hola " + strcharinfo(0) + ",";
	mes "quieres que te vincule con un espiritu ancestral?";
	mes "El ritual cuesta " + .price + " zeny.";
	mes "Dura unos " + (.duration / 60000) + " minutos.";
	next;

	switch (select("Si, vinculame", "Que espiritu me espera?", "No, gracias")) {
	case 3:
		mes .@name$;
		mes "Como quieras. Los espiritus esperaran.";
		close;
	case 2:
		.@Spirit = callsub(S_ResolveSpirit);
		if (!.@Spirit) {
			mes .@name$;
			mes "No encuentro un espiritu para tu camino.";
			mes "BaseJob: " + BaseJob + " (" + jobname(BaseJob) + ")";
			close;
		}
		callsub S_SpiritHint;
		close;
	case 1:
		break;
	}

	.@Spirit = callsub(S_ResolveSpirit);
	if (!.@Spirit) {
		mes .@name$;
		mes "No encuentro un espiritu para tu camino.";
		mes "BaseJob: " + BaseJob + " (" + jobname(BaseJob) + ")";
		close;
	}

	if (Zeny < .price) {
		mes .@name$;
		mes "Necesitas " + .price + " zeny para el ritual.";
		close;
	}

	Zeny -= .price;
	sc_start2 SC_SPIRIT, .duration, 5, .@Spirit;
	skilleffect .@Spirit, 5;
	callsub S_SpiritSuccess, .@Spirit;
	close;

S_ResolveSpirit:
	.@Spirit = 0;
	switch (BaseJob) {
	case Job_Alchemist:      .@Spirit = SL_ALCHEMIST; break;
	case Job_Monk:           .@Spirit = SL_MONK; break;
	case Job_Star_Gladiator: .@Spirit = SL_STAR; break;
	case Job_Sage:           .@Spirit = SL_SAGE; break;
	case Job_Crusader:       .@Spirit = SL_CRUSADER; break;
	case Job_SuperNovice:    .@Spirit = SL_SUPERNOVICE; break;
	case Job_Knight:         .@Spirit = SL_KNIGHT; break;
	case Job_Wizard:         .@Spirit = SL_WIZARD; break;
	case Job_Priest:         .@Spirit = SL_PRIEST; break;
	case Job_Bard:
	case Job_Dancer:         .@Spirit = SL_BARDDANCER; break;
	case Job_Rogue:          .@Spirit = SL_ROGUE; break;
	case Job_Assassin:       .@Spirit = SL_ASSASIN; break;
	case Job_Blacksmith:     .@Spirit = SL_BLACKSMITH; break;
	case Job_Hunter:         .@Spirit = SL_HUNTER; break;
	case Job_Soul_Linker:    .@Spirit = SL_SOULLINKER; break;
	case Job_Ninja:          .@Spirit = SL_NINJA; break;
	case Job_Gunslinger:     .@Spirit = SL_GUNNER; break;
	default:
		if (Upper == 2 && BaseLevel < 70)
			.@Spirit = 494; // SL_HIGH
		break;
	}
	return .@Spirit;

S_SpiritHint:
	mes .@name$;
	mes "Cierra los ojos... dejame escuchar tu alma.";
	next;
	mes .@name$;
	switch (BaseJob) {
	case Job_Alchemist:
		mes "Huelo brebajes y vidrio. Un maestro de frascos";
		mes "que convierte la alquimia en auxilio y furia contenida.";
		break;
	case Job_Assassin:
		mes "Sombras que se pliegan dos veces antes del ultimo suspiro.";
		mes "Un asesino cuyo golpe no perdona al que vacila.";
		break;
	case Job_Bard:
	case Job_Dancer:
		mes "Oigo una melodia que vuelve sobre quien la canta.";
		mes "Bardo y bailarina comparten un mismo latido en el escenario.";
		break;
	case Job_Blacksmith:
		mes "El martillo late como un segundo corazon.";
		mes "Un herrero capaz de despertar el acero con adrenalina pura.";
		break;
	case Job_Crusader:
		mes "Un escudo que vuelve como jabalina de fe.";
		mes "Cruzado cuya devocion golpea dos veces con la misma furia.";
		break;
	case Job_Hunter:
		mes "Rastros en el barro y flechas que conocen la debilidad de toda bestia.";
		mes "Un cazador que lee el pulso del mundo salvaje.";
		break;
	case Job_Knight:
		mes "Una mano firme y una hoja que no conoce la duda.";
		mes "Caballero cuya velocidad nace del temple, no del miedo.";
		break;
	case Job_Monk:
		mes "Punos que agitan el aire en cruce de tormenta.";
		mes "Monje cuya furia interior alimenta cada combo.";
		break;
	case Job_Priest:
		mes "Luz sagrada que arde con fe inquebrantable.";
		mes "Sacerdote cuya oracion pesa como el juicio del cielo.";
		break;
	case Job_Rogue:
		mes "Huellas que se borran entre la niebla.";
		mes "Picaro que camina donde otros solo ven muros.";
		break;
	case Job_Sage:
		mes "Paginas que se escriben solas en el instante preciso.";
		mes "Sabio que invoca conocimiento cuando la mente lo exige.";
		break;
	case Job_Soul_Linker:
		mes "Hilos invisibles que unen almas distintas.";
		mes "Vinculador que comparte poder sin reservarlo solo para si.";
		break;
	case Job_SuperNovice:
		mes "Un destino improbable escrito con tinta de estrellas.";
		mes "Novicio eterno al que el mundo aun no termina de comprender.";
		break;
	case Job_Star_Gladiator:
		mes "Constelaciones que arden en los nudillos de un guerrero.";
		mes "Maestro marcial que canaliza el cosmos en un solo golpe.";
		break;
	case Job_Wizard:
		mes "Gemas que obedecen sin quebrarse.";
		mes "Mago que domina el coste de lo arcano como quien cuenta monedas.";
		break;
	case Job_Ninja:
		mes "Hojas que caen antes de que el viento avise.";
		mes "Sombra de Izlude que corta sin pedir permiso.";
		break;
	case Job_Gunslinger:
		mes "Polvora, precision y humo que se disipa tarde.";
		mes "Artillero que mide la vida en disparos certeros.";
		break;
	default:
		if (Upper == 2 && BaseLevel < 70) {
			mes "Veo un hilo de luz que aun no ha alcanzado su cima.";
			mes "Un eco de quien fuiste, dispuesto a prestarte su vigor renacido.";
		} else {
			mes "Algo se agita en tu camino, pero aun no logro nombrarlo.";
		}
		break;
	}
	next;
	mes .@name$;
	mes "Si deseas el vinculo, vuelve cuando estes listo.";
	return;

S_SpiritSuccess:
	mes .@name$;
	if (getarg(0) == 494) {
		mes "Has obtenido el ^0055FFEspiritu del Renacimiento^000000.";
		mes "Siente como un eco de tu yo anterior te presta su fuerza.";
	} else {
		mes "Has obtenido el espiritu de un gran ^00CCFF" + jobname(BaseJob) + "^000000.";
		mes "Que te guie en la batalla.";
	}
	return;

OnInit:
	.duration = 1200000; // 20 min; futuro: quest puede subir este valor
	.price = 20000;
	end;
}

comodo,182,155,4	duplicate(Kessinger)	Kessinger#comodo	4_M_ROTERT
