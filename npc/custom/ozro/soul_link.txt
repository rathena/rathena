	mes "Hola " + strcharinfo(0) + ".";
	next;

	// --- Debug: variables del personaje ---

	mes "[Debug] Class: " + Class + " (" + jobname(Class) + ")";

	next;

	// --- Resolucion de espiritu (3 pasos) ---

	// Paso 1: job base 2-1 / 2-2 (funciona en la mayoria de 1st/2nd)
	switch (BaseJob) {
	case Job_Alchemist:      .@Spirit = SL_ALCHEMIST; break;
	case Job_Monk:           .@Spirit = SL_MONK; break;
	case Job_Star_Gladiator: .@Spirit = SL_STAR; break;
	case Job_Sage:           .@Spirit = SL_SAGE; break;
	case Job_Crusader:       .@Spirit = SL_CRUSADER; break;
	case Job_SuperNovice:    .@Spirit = SL_SUPERNOVICE; break;
	case Job_Knight:         .@Spirit = SL_KNIGHT; break;
	case Job_Wizard:         .@Spirit = SL_WIZARD; break;
	case Job_Priest:         .@Spirit = SL_PRIEST; break;
	case Job_Bard:
	case Job_Dancer:         .@Spirit = SL_BARDDANCER; break;
	case Job_Rogue:          .@Spirit = SL_ROGUE; break;
	case Job_Assassin:       .@Spirit = SL_ASSASIN; break;
	case Job_Blacksmith:     .@Spirit = SL_BLACKSMITH; break;
	case Job_Hunter:         .@Spirit = SL_HUNTER; break;
	case Job_Soul_Linker:    .@Spirit = SL_SOULLINKER; break;
	case Job_Ninja:          .@Spirit = SL_NINJA; break;
	case Job_Gunslinger:     .@Spirit = SL_GUNNER; break;
	default:
		if (Upper == 2 && BaseLevel < 70)
			.@Spirit = SL_HIGH;
		break;
	}

	// Paso 2: Class actual (renewal / 3rd / 4th)
	if (!.@Spirit) {
		switch (Class) {
		case Job_Creator:
		case Job_Genetic:
		case Job_Genetic_T:
		case Job_Genetic_2ND:
		case Job_Biolo:
		case Job_Baby_Alchemist:     .@Spirit = SL_ALCHEMIST; break;
		case Job_Lord_Knight:
		case Job_Rune_Knight:
		case Job_Rune_Knight_T:
		case Job_Dragon_Knight:
		case Job_Baby_Rune_Knight:   .@Spirit = SL_KNIGHT; break;
		case Job_High_Wizard:
		case Job_Warlock:
		case Job_Warlock_T:
		case Job_Arch_Mage:            .@Spirit = SL_WIZARD; break;
		case Job_Sniper:
		case Job_Ranger:
		case Job_Ranger_T:
		case Job_Windhawk:             .@Spirit = SL_HUNTER; break;

		case Job_Whitesmith:
		case Job_Mechanic:
		case Job_Mechanic_T:
		case Job_Meister:              .@Spirit = SL_BLACKSMITH; break;
		case Job_Assassin_Cross:
		case Job_Guillotine_Cross:
		case Job_Guillotine_Cross_T:
		case Job_Shadow_Cross:         .@Spirit = SL_ASSASIN; break;
		case Job_Paladin:
		case Job_Royal_Guard:
		case Job_Royal_Guard_T:
		case Job_Imperial_Guard:       .@Spirit = SL_CRUSADER; break;
		case Job_Professor:
		case Job_Sorcerer:
		case Job_Sorcerer_T:
		case Job_Elemental_Master:       .@Spirit = SL_SAGE; break;
		case Job_Champion:
		case Job_Sura:
		case Job_Sura_T:
		case Job_Inquisitor:           .@Spirit = SL_MONK; break;
		case Job_Clown:
		case Job_Gypsy:
		case Job_Minstrel:
		case Job_Wanderer:
		case Job_Minstrel_T:
		case Job_Wanderer_T:
		case Job_Troubadour:
		case Job_Trouvere:             .@Spirit = SL_BARDDANCER; break;
		case Job_Stalker:
		case Job_Shadow_Chaser:
		case Job_Shadow_Chaser_T:
		case Job_Abyss_Chaser:         .@Spirit = SL_ROGUE; break;
		case Job_Star_Emperor:
		case Job_Star_Emperor2:
		case Job_Baby_Star_Gladiator:  .@Spirit = SL_STAR; break;
		case Job_Night_Watch:
		case Job_Rebellion:            .@Spirit = SL_GUNNER; break;
		case Job_Shinkiro:
		case Job_Shiranui:             .@Spirit = SL_NINJA; break;
		case Job_Death_Knight:         .@Spirit = SL_DEATHKNIGHT; break;
		case Job_Dark_Collector:       .@Spirit = SL_COLLECTOR; break;
		}
	}

	// Paso 3: eaclass mask (ultimo recurso)
	if (!.@Spirit) {
		switch (eaclass() & EAJ_UPPERMASK) {
		case EAJ_KNIGHT:
		case EAJ_LORD_KNIGHT:
		case EAJ_RUNE_KNIGHT:          .@Spirit = SL_KNIGHT; break;
		case EAJ_WIZARD:
		case EAJ_HIGH_WIZARD:
		case EAJ_WARLOCK:              .@Spirit = SL_WIZARD; break;
		case EAJ_HUNTER:
		case EAJ_SNIPER:
		case EAJ_RANGER:               .@Spirit = SL_HUNTER; break;
		case EAJ_PRIEST:
		case EAJ_HIGH_PRIEST:
		case EAJ_ARCH_BISHOP:          .@Spirit = SL_PRIEST; break;
		case EAJ_BLACKSMITH:
		case EAJ_WHITESMITH:
		case EAJ_MECHANIC:             .@Spirit = SL_BLACKSMITH; break;
		case EAJ_ASSASSIN:
		case EAJ_ASSASSIN_CROSS:
		case EAJ_GUILLOTINE_CROSS:     .@Spirit = SL_ASSASIN; break;
		case EAJ_CRUSADER:
		case EAJ_PALADIN:
		case EAJ_ROYAL_GUARD:           .@Spirit = SL_CRUSADER; break;
		case EAJ_SAGE:
		case EAJ_PROFESSOR:
		case EAJ_SORCERER:             .@Spirit = SL_SAGE; break;
		case EAJ_MONK:
		case EAJ_CHAMPION:
		case EAJ_SURA:                 .@Spirit = SL_MONK; break;

		case EAJ_CLOWNGYPSY:
		case EAJ_MINSTRELWANDERER:     .@Spirit = SL_BARDDANCER; break;
		case EAJ_ROGUE:
		case EAJ_STALKER:              .@Spirit = SL_ROGUE; break;
		case EAJ_ALCHEMIST:
		case EAJ_CREATOR:
		case EAJ_GENETIC:
		case EAJ_GENETIC_T:
		case EAJ_BABY_ALCHEMIST:       .@Spirit = SL_ALCHEMIST; break;
		case EAJ_SUPER_NOVICE:         .@Spirit = SL_SUPERNOVICE; break;
		case EAJ_STAR_GLADIATOR:
		case EAJ_STAR_EMPEROR:         .@Spirit = SL_STAR; break;
		case EAJ_SOUL_LINKER:          .@Spirit = SL_SOULLINKER; break;
		case EAJ_NINJA:                .@Spirit = SL_NINJA; break;
		case EAJ_GUNSLINGER:
		case EAJ_REBELLION:            .@Spirit = SL_GUNNER; break;
		case EAJ_DEATH_KNIGHT:         .@Spirit = SL_DEATHKNIGHT; break;
		case EAJ_DARK_COLLECTOR:       .@Spirit = SL_COLLECTOR; break;
		}
	}

	// Nombre del espiritu en espanol (para probar el switch)
	.@spiritName$ = "Ninguno";
	switch (.@Spirit) {
	case SL_HIGH:          .@spiritName$ = "Renacimiento"; break;
	case SL_ALCHEMIST:     .@spiritName$ = "Alquimista"; break;
	case SL_ASSASIN:       .@spiritName$ = "Asesino"; break;
	case SL_BARDDANCER:    .@spiritName$ = "Bardo o Bailarin"; break;
	case SL_BLACKSMITH:    .@spiritName$ = "Herrero"; break;
	case SL_CRUSADER:      .@spiritName$ = "Cruzado"; break;
	case SL_HUNTER:        .@spiritName$ = "Cazador"; break;
	case SL_KNIGHT:        .@spiritName$ = "Caballero"; break;
	case SL_MONK:          .@spiritName$ = "Monje"; break;
	case SL_PRIEST:        .@spiritName$ = "Sacerdote"; break;
	case SL_ROGUE:         .@spiritName$ = "Picaro"; break;
	case SL_SAGE:          .@spiritName$ = "Sabio"; break;
	case SL_SOULLINKER:    .@spiritName$ = "Vinculador de Almas"; break;
	case SL_SUPERNOVICE:   .@spiritName$ = "Super Novicio"; break;
	case SL_STAR:          .@spiritName$ = "Maestro Taekwon"; break;
	case SL_WIZARD:        .@spiritName$ = "Mago"; break;
	case SL_NINJA:         .@spiritName$ = "Ninja"; break;
	case SL_GUNNER:        .@spiritName$ = "Pistolero"; break;
	case SL_DEATHKNIGHT:   .@spiritName$ = "Caballero de la Muerte"; break;
	case SL_COLLECTOR:     .@spiritName$ = "Coleccionista Oscuro"; break;
	}

	mes .@name$;
	if (!.@Spirit)
		mes "No detecte espiritu compatible.";
	else
		mes "Espiritu detectado: ^00CCFF" + .@spiritName$ + "^000000 (ID " + .@Spirit + ")";
	close;

	/*

		return;
	*/

	.duration = 1200000; // 20 min; futuro: quest puede subir este valor
	.price = 20000;
	end;
}
