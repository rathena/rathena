//===== OzRo Script ==========================================
//= Kessinger, Vinculador de Almas
//===== Description: =========================================
//= Soul Link de pago para jobs compatibles.
//============================================================

yuno,145,185,4	script	Kessinger	4_M_ROTERT,{

	.@name$ = "[ ^0000FFKessinger, Vinculador de Almas^000000 ]";

	mes .@name$;
	mes "Hola " + strcharinfo(0) + ",";
	mes "quieres que te vincule con un espiritu ancestral?";
	mes "El ritual cuesta " + .price + " zeny.";
	next;

	if (select("Si, vinculame", "No, gracias") == 2) {
		mes .@name$;
		mes "Como quieras. Los espiritus esperaran.";
		close;
	}

	if (Zeny < .price) {
		mes .@name$;
		mes "Necesitas " + .price + " zeny para el ritual.";
		close;
	}

	.@Spirit = 0;
	switch (BaseJob) {
	case Job_Alchemist:
		.@Spirit = SL_ALCHEMIST;
		break;
	case Job_Monk:
		.@Spirit = SL_MONK;
		break;
	case Job_Star_Gladiator:
		.@Spirit = SL_STAR;
		break;
	case Job_Sage:
		.@Spirit = SL_SAGE;
		break;
	case Job_Crusader:
		.@Spirit = SL_CRUSADER;
		break;
	case Job_SuperNovice:
		.@Spirit = SL_SUPERNOVICE;
		break;
	case Job_Knight:
		.@Spirit = SL_KNIGHT;
		break;
	case Job_Wizard:
		.@Spirit = SL_WIZARD;
		break;
	case Job_Priest:
		.@Spirit = SL_PRIEST;
		break;
	case Job_Bard:
	case Job_Dancer:
		.@Spirit = SL_BARDDANCER;
		break;
	case Job_Rogue:
		.@Spirit = SL_ROGUE;
		break;
	case Job_Assassin:
		.@Spirit = SL_ASSASIN;
		break;
	case Job_Blacksmith:
		.@Spirit = SL_BLACKSMITH;
		break;
	case Job_Hunter:
		.@Spirit = SL_HUNTER;
		break;
	case Job_Soul_Linker:
		.@Spirit = SL_SOULLINKER;
		break;
	case Job_Ninja:
		.@Spirit = SL_NINJA;
		break;
	case Job_Gunslinger:
		.@Spirit = SL_GUNNER;
		break;
	default:
		if (Upper == 2 && BaseLevel < 70)
			.@Spirit = SL_HIGH;
		break;
	}

	if (!.@Spirit) {
		mes .@name$;
		mes "Lo siento. No encuentro un espiritu compatible con tu camino.";
		close;
	}

	Zeny -= .price;
	sc_start2 SC_SOULLINK, .duration, 5, .@Spirit;
	skilleffect .@Spirit, 5;

	mes .@name$;
	if (.@Spirit == SL_HIGH)
		mes "Siente el ^0055FFEspiritu del Renacimiento^000000 fluir en ti.";
	else
		mes "El vinculo esta hecho. Que el espiritu te guie en batalla.";
	close;

OnInit:
	.duration = 1200000;
	.price = 20000;
	end;
}

prontera,161,185,4	duplicate(Kessinger)	Kessinger#protera	4_M_ROTERT
morocc,148,106,4	duplicate(Kessinger)	Kessinger#morroc	4_M_ROTERT
comodo,182,155,4	duplicate(Kessinger)	Kessinger#comodo	4_M_ROTERT
geffen,125,72,4	duplicate(Kessinger)	Kessinger#geffen	4_M_ROTERT
