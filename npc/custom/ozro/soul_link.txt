yuno,145,185,4	script	Kessinger	4_M_ROTERT,{

	.@name$ = "[ ^0000FFKessinger, Vinculador de Almas^000000 ]";

	mes .@name$;
	mes "Hola " + strcharinfo(0) + ",";
	mes "quieres que te vincule con un espiritu ancestral?";
	mes "El ritual cuesta " + .price + " zeny.";
	mes "Dura unos " + (.duration / 60000) + " minutos.";
	next;

	switch (select("Si, vinculame", "Que espiritu me espera?", "No, gracias")) {
	case 3:
		mes .@name$;
		mes "Como quieras. Los espiritus esperaran.";
		close;
	case 2:
		.@Spirit = callsub(S_ResolveSpirit);
		if (!.@Spirit) {
			callsub S_NoSpirit;
			close;
		}
		callsub S_SpiritHint, .@Spirit;
		close;
	case 1:
		break;
	}

	.@Spirit = callsub(S_ResolveSpirit);
	if (!.@Spirit) {
		callsub S_NoSpirit;
		close;
	}

	if (Zeny < .price) {
		mes .@name$;
		mes "Necesitas " + .price + " zeny para el ritual.";
		close;
	}

	Zeny -= .price;
	sc_start2 SC_SPIRIT, .duration, 5, .@Spirit;
	skilleffect .@Spirit, 5;

	callsub S_SpiritSuccess, .@Spirit;
	close;

S_ResolveSpirit:
	.@Spirit = 0;

	switch (BaseJob) {
	case Job_Alchemist:     .@Spirit = SL_ALCHEMIST; break;
	case Job_Monk:          .@Spirit = SL_MONK; break;
	case Job_Star_Gladiator:.@Spirit = SL_STAR; break;
	case Job_Sage:          .@Spirit = SL_SAGE; break;
	case Job_Crusader:      .@Spirit = SL_CRUSADER; break;
	case Job_SuperNovice:   .@Spirit = SL_SUPERNOVICE; break;
	case Job_Knight:        .@Spirit = SL_KNIGHT; break;
	case Job_Wizard:        .@Spirit = SL_WIZARD; break;
	case Job_Priest:        .@Spirit = SL_PRIEST; break;
	case Job_Bard:
	case Job_Dancer:        .@Spirit = SL_BARDDANCER; break;
	case Job_Rogue:         .@Spirit = SL_ROGUE; break;
	case Job_Assassin:      .@Spirit = SL_ASSASIN; break;
	case Job_Blacksmith:    .@Spirit = SL_BLACKSMITH; break;
	case Job_Hunter:        .@Spirit = SL_HUNTER; break;
	case Job_Soul_Linker:   .@Spirit = SL_SOULLINKER; break;
	case Job_Ninja:         .@Spirit = SL_NINJA; break;
	case Job_Gunslinger:    .@Spirit = SL_GUNNER; break;
	default:
		if (Upper == 2 && BaseLevel < 70)
			.@Spirit = SL_HIGH;
		break;
	}

	if (!.@Spirit) {
		switch (Class) {
		case Job_Creator:
		case Job_Genetic:
		case Job_Genetic_T:
		case Job_Genetic_2ND:
		case Job_Biolo:
		case Job_Baby_Alchemist:        .@Spirit = SL_ALCHEMIST; break;
		case Job_Lord_Knight:
		case Job_Rune_Knight:
		case Job_Rune_Knight_T:
		case Job_Dragon_Knight:
		case Job_Baby_Rune_Knight:      .@Spirit = SL_KNIGHT; break;
		case Job_High_Wizard:
		case Job_Warlock:
		case Job_Warlock_T:
		case Job_Arch_Mage:               .@Spirit = SL_WIZARD; break;
		case Job_Sniper:
		case Job_Ranger:
		case Job_Ranger_T:
		case Job_Windhawk:                .@Spirit = SL_HUNTER; break;
		case Job_High_Priest:
		case Job_Arch_Bishop:
		case Job_Arch_Bishop_T:
		case Job_Cardinal:                .@Spirit = SL_PRIEST; break;
		case Job_Whitesmith:
		case Job_Mechanic:
		case Job_Mechanic_T:
		case Job_Meister:                 .@Spirit = SL_BLACKSMITH; break;
		case Job_Assassin_Cross:
		case Job_Guillotine_Cross:
		case Job_Guillotine_Cross_T:
		case Job_Shadow_Cross:            .@Spirit = SL_ASSASIN; break;
		case Job_Paladin:
		case Job_Royal_Guard:
		case Job_Royal_Guard_T:
		case Job_Imperial_Guard:          .@Spirit = SL_CRUSADER; break;
		case Job_Professor:
		case Job_Sorcerer:
		case Job_Sorcerer_T:
		case Job_Elemental_Master:          .@Spirit = SL_SAGE; break;
		case Job_Champion:
		case Job_Sura:
		case Job_Sura_T:
		case Job_Inquisitor:              .@Spirit = SL_MONK; break;
		case Job_Clown:
		case Job_Gypsy:
		case Job_Minstrel:
		case Job_Wanderer:
		case Job_Minstrel_T:
		case Job_Wanderer_T:
		case Job_Troubadour:
		case Job_Trouvere:                .@Spirit = SL_BARDDANCER; break;
		case Job_Stalker:
		case Job_Shadow_Chaser:
		case Job_Shadow_Chaser_T:
		case Job_Abyss_Chaser:            .@Spirit = SL_ROGUE; break;
		case Job_Star_Emperor:
		case Job_Star_Emperor2:
		case Job_Baby_Star_Gladiator:     .@Spirit = SL_STAR; break;
		case Job_Night_Watch:
		case Job_Rebellion:               .@Spirit = SL_GUNNER; break;
		case Job_Shinkiro:
		case Job_Shiranui:                .@Spirit = SL_NINJA; break;
		case Job_Death_Knight:            .@Spirit = SL_DEATHKNIGHT; break;
		case Job_Dark_Collector:          .@Spirit = SL_COLLECTOR; break;
		}
	}

	if (!.@Spirit) {
		.@eac = eaclass();
		switch (.@eac & EAJ_UPPERMASK) {
		case EAJ_KNIGHT:
		case EAJ_LORD_KNIGHT:
		case EAJ_RUNE_KNIGHT:             .@Spirit = SL_KNIGHT; break;
		case EAJ_WIZARD:
		case EAJ_HIGH_WIZARD:
		case EAJ_WARLOCK:                 .@Spirit = SL_WIZARD; break;
		case EAJ_HUNTER:
		case EAJ_SNIPER:
		case EAJ_RANGER:                  .@Spirit = SL_HUNTER; break;
		case EAJ_PRIEST:
		case EAJ_HIGH_PRIEST:
		case EAJ_ARCH_BISHOP:             .@Spirit = SL_PRIEST; break;
		case EAJ_BLACKSMITH:
		case EAJ_WHITESMITH:
		case EAJ_MECHANIC:                .@Spirit = SL_BLACKSMITH; break;
		case EAJ_ASSASSIN:
		case EAJ_ASSASSIN_CROSS:
		case EAJ_GUILLOTINE_CROSS:        .@Spirit = SL_ASSASIN; break;
		case EAJ_CRUSADER:
		case EAJ_PALADIN:
		case EAJ_ROYAL_GUARD:              .@Spirit = SL_CRUSADER; break;
		case EAJ_SAGE:
		case EAJ_PROFESSOR:
		case EAJ_SORCERER:                .@Spirit = SL_SAGE; break;
		case EAJ_MONK:
		case EAJ_CHAMPION:
		case EAJ_SURA:                    .@Spirit = SL_MONK; break;
		case EAJ_BARDDANCER:
		case EAJ_CLOWNGYPSY:
		case EAJ_MINSTRELWANDERER:        .@Spirit = SL_BARDDANCER; break;
		case EAJ_ROGUE:
		case EAJ_STALKER:                 .@Spirit = SL_ROGUE; break;
		case EAJ_ALCHEMIST:
		case EAJ_CREATOR:
		case EAJ_GENETIC:
		case EAJ_GENETIC_T:
		case EAJ_BABY_ALCHEMIST:          .@Spirit = SL_ALCHEMIST; break;
		case EAJ_SUPER_NOVICE:            .@Spirit = SL_SUPERNOVICE; break;
		case EAJ_STAR_GLADIATOR:
		case EAJ_STAR_EMPEROR:            .@Spirit = SL_STAR; break;
		case EAJ_SOUL_LINKER:             .@Spirit = SL_SOULLINKER; break;
		case EAJ_NINJA:                   .@Spirit = SL_NINJA; break;
		case EAJ_GUNSLINGER:
		case EAJ_REBELLION:               .@Spirit = SL_GUNNER; break;
		case EAJ_DEATH_KNIGHT:            .@Spirit = SL_DEATHKNIGHT; break;
		case EAJ_DARK_COLLECTOR:          .@Spirit = SL_COLLECTOR; break;
		}
	}

	return .@Spirit;

S_NoSpirit:
	mes .@name$;
	mes "Lo siento. No encuentro un espiritu compatible con tu camino.";
	if (getgmlevel() > 0) {
		mes "Class: " + Class + " (" + jobname(Class) + ")";
		mes "BaseJob: " + BaseJob + " (" + jobname(BaseJob) + ")";
		mes "eaclass: " + eaclass();
	}
	return;

S_SpiritHint:
	.@Spirit = getarg(0);
	mes .@name$;
	mes "Cierra los ojos... dejame escuchar tu alma.";
	next;
	mes .@name$;
	switch (.@Spirit) {
	case SL_HIGH:
		mes "Veo un hilo de luz que aun no ha alcanzado su cima.";
		mes "Un eco de quien fuiste, dispuesto a prestarte su vigor renacido.";
		break;
	case SL_ALCHEMIST:
		mes "Huelo brebajes y vidrio. Un maestro de frascos";
		mes "que convierte la alquimia en auxilio y furia contenida.";
		break;
	case SL_ASSASIN:
		mes "Sombras que se pliegan dos veces antes del ultimo suspiro.";
		mes "Un asesino cuyo golpe no perdona al que vacila.";
		break;
	case SL_BARDDANCER:
		mes "Oigo una melodia que vuelve sobre quien la canta.";
		mes "Bardo y bailarina comparten un mismo latido en el escenario.";
		break;
	case SL_BLACKSMITH:
		mes "El martillo late como un segundo corazon.";
		mes "Un herrero capaz de despertar el acero con adrenalina pura.";
		break;
	case SL_CRUSADER:
		mes "Un escudo que vuelve como jabalina de fe.";
		mes "Cruzado cuya devocion golpea dos veces con la misma furia.";
		break;
	case SL_HUNTER:
		mes "Rastros en el barro y flechas que conocen la debilidad de toda bestia.";
		mes "Un cazador que lee el pulso del mundo salvaje.";
		break;
	case SL_KNIGHT:
		mes "Una mano firme y una hoja que no conoce la duda.";
		mes "Caballero cuya velocidad nace del temple, no del miedo.";
		break;
	case SL_MONK:
		mes "Punos que agitan el aire en cruce de tormenta.";
		mes "Monje cuya furia interior alimenta cada combo.";
		break;
	case SL_PRIEST:
		mes "Luz sagrada que arde con fe inquebrantable.";
		mes "Sacerdote cuya oracion pesa como el juicio del cielo.";
		break;
	case SL_ROGUE:
		mes "Huellas que se borran entre la niebla.";
		mes "Picaro que camina donde otros solo ven muros.";
		break;
	case SL_SAGE:
		mes "Paginas que se escriben solas en el instante preciso.";
		mes "Sabio que invoca conocimiento cuando la mente lo exige.";
		break;
	case SL_SOULLINKER:
		mes "Hilos invisibles que unen almas distintas.";
		mes "Vinculador que comparte poder sin reservarlo solo para si.";
		break;
	case SL_SUPERNOVICE:
		mes "Un destino improbable escrito con tinta de estrellas.";
		mes "Novicio eterno al que el mundo aun no termina de comprender.";
		break;
	case SL_STAR:
		mes "Constelaciones que arden en los nudillos de un guerrero.";
		mes "Maestro marcial que canaliza el cosmos en un solo golpe.";
		break;
	case SL_WIZARD:
		mes "Gemas que obedecen sin quebrarse.";
		mes "Mago que domina el coste de lo arcano como quien cuenta monedas.";
		break;
	case SL_NINJA:
		mes "Hojas que caen antes de que el viento avise.";
		mes "Sombra de Izlude que corta sin pedir permiso.";
		break;
	case SL_GUNNER:
		mes "Polvora, precision y humo que se disipa tarde.";
		mes "Artillero que mide la vida en disparos certeros.";
		break;
	case SL_DEATHKNIGHT:
		mes "Armadura oscura y un juramento roto en el abismo.";
		mes "Caballero maldito que aun no ha olvidado como luchar.";
		break;
	case SL_COLLECTOR:
		mes "Ecos de almas perdidas que aun buscan rumbo.";
		mes "Recolector que escucha lo que el mundo prefiere olvidar.";
		break;
	default:
		mes "Algo se agita en tu camino, pero aun no logro nombrarlo.";
		break;
	}
	next;
	mes .@name$;
	mes "Si deseas el vinculo, vuelve cuando estes listo.";
	return;

S_SpiritSuccess:
	.@Spirit = getarg(0);
	mes .@name$;
	switch (.@Spirit) {
	case SL_HIGH:
		mes "Has obtenido el ^0055FFEspiritu del Renacimiento^000000.";
		mes "Siente como un eco de tu yo anterior te presta su fuerza.";
		break;
	case SL_ALCHEMIST:
		mes "Has obtenido el espiritu de un gran ^00CCFFAlchemist^000000.";
		mes "Que sus brebajes te sostengan en la batalla.";
		break;
	case SL_ASSASIN:
		mes "Has obtenido el espiritu de un gran ^990099Assassin^000000.";
		mes "Que tus golpes lleguen antes que el miedo.";
		break;
	case SL_BARDDANCER:
		mes "Has obtenido el espiritu de un gran ^FF66CCBardo o Bailarin^000000.";
		mes "Que tu arte vuelva sobre ti como bendicion.";
		break;
	case SL_BLACKSMITH:
		mes "Has obtenido el espiritu de un gran ^AAAAAABlacksmith^000000.";
		mes "Que el acero obedezca tu ritmo.";
		break;
	case SL_CRUSADER:
		mes "Has obtenido el espiritu de un gran ^FFFF00Crusader^000000.";
		mes "Que tu fe golpee con el peso del escudo.";
		break;
	case SL_HUNTER:
		mes "Has obtenido el espiritu de un gran ^77FF77Hunter^000000.";
		mes "Que ninguna bestia escape a tu mira.";
		break;
	case SL_KNIGHT:
		mes "Has obtenido el espiritu de un gran ^7777FFKnight^000000.";
		mes "Que tu hoja sea extension de tu voluntad.";
		break;
	case SL_MONK:
		mes "Has obtenido el espiritu de un gran ^FFAA00Monk^000000.";
		mes "Que cada puno lleve la calma y la tormenta a la vez.";
		break;
	case SL_PRIEST:
		mes "Has obtenido el espiritu de un gran ^FFFFFFPriest^000000.";
		mes "Que la luz te acompane sin pedir permiso.";
		break;
	case SL_ROGUE:
		mes "Has obtenido el espiritu de un gran ^888888Rogue^000000.";
		mes "Que la sombra te cubra cuando mas lo necesites.";

	case SL_NINJA:

		mes "Has obtenido el espiritu de un gran ^CC9966Gunslinger^000000.";

	.duration = 1200000; // 20 min; futuro: quest puede subir este valor
