// ==========================================
// 1. COMBINED EQUIPMENT VENDOR
// ==========================================
prontera,147,242,5	script	Equipment Merchant#p_market	4_M_MERCHANT,{
	mes "[Equipment Merchant]";
	mes "Welcome! What type of combat gear are you looking for today?";
	next;
	switch(select("Armor & Footwear:Melee Weapons:Ranged Weapons & Instruments:Books, Staffs, & Rods:Quit")) {
		case 1:
			switch(select("Body Armor:Shields:Footwear")) {
				case 1: callshop "p_armor",1; break;
				case 2: callshop "p_shield",1; break;
				case 3: callshop "p_footwear",1; break;
			}
			break;
		case 2:
			switch(select("Maces & Knuckles:Daggers & Katars:One-Handed Swords:Two-Handed Swords:Spears & Axes")) {
				case 1: if(select("Maces:Knuckles")==1) callshop "p_mace",1; else callshop "p_knuckle",1; break;
				case 2: if(select("Daggers:Katars")==1) callshop "p_dagger",1; else callshop "p_katar",1; break;
				case 3: callshop "p_1hsword",1; break;
				case 4: callshop "p_2hsword",1; break;
				case 5: if(select("Spears:Axes")==1) callshop "p_spear",1; else callshop "p_axe",1; break;
			}
			break;
		case 3:
			switch(select("Whips:Instruments:Guns:Bows & Shurikens")) {
				case 1: callshop "p_whip",1; break;
				case 2: callshop "p_instrument",1; break;
				case 3: callshop "p_gun",1; break;
				case 4: if(select("Bows:Fuuma Shurikens")==1) callshop "p_bow",1; else callshop "p_shuriken",1; break;
			}
			break;
		case 4:
			if(select("Books:Staffs & Rods")==1) callshop "p_book",1; else callshop "p_staff",1; break;
		case 5:
			close;
	}
	end;
}

		case 3: callshop "p_up_hg",1; break;
		case 4: 
			switch(select("Headgear Page 01:Headgear Page 02:Headgear Page 03")) {
				case 1: callshop "p_hg01",1; break;
				case 2: callshop "p_hg02",1; break;
				case 3: callshop "p_hg03",1; break;
			}
			break;
		case 5:
			if(select("Headgear Page 04:Headgear Page 05")==1) callshop "p_hg04",1; else callshop "p_hg05",1; break;
		case 6:
			close;
	}
	end;
}
