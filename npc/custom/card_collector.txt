//===== Ragnarok Offline =====================================
//= Card Collector
//===== Description: =========================================
//= Buys standalone card items for account-wide Cash Points.
//= Offers scale with the canonical source monster level:
//= Normal 1-50: 250-1000; 51-100: 2000-4000; 101+: 7000-15000;
//= Miniboss: 7000-20000; MVP: 20000-100000.
//= Account-wide loyalty counters reward repeat Normal, Miniboss,
//= and MVP card sales. Multiple milestones roll independently.
//============================================================

paramk,41,159,4	script	Black Market Boss#offline	4_M_MANAGER,{
	mes "[Black Market Boss]";
	mes "So, you brought me cards.";
	mes "Good. Every card that changes hands in this market eventually reaches my table.";
	mes "I pay in ^0055FFCash Points^000000, not Zeny.";
	mes "My offer depends on the monster behind each card.";
	next;
	switch (select("Sell cards.:Check points and loyalty.:Leave.")) {
	case 1:
		getinventorylist;
		.@card_count = 0;
		for (.@i = 0; .@i < @inventorylist_count; ++.@i) {
			.@item_id = @inventorylist_id[.@i];
			if (getiteminfo(.@item_id, ITEMINFO_TYPE) != IT_CARD ||
				getiteminfo(.@item_id, ITEMINFO_SUBTYPE) != CARD_NORMAL ||
				callsub(S_GetOffer, .@item_id) <= 0)
				continue;
			if (inarray(.@card_ids[0], .@item_id) == -1)
				.@card_ids[.@card_count++] = .@item_id;
		}

		if (!.@card_count) {
			mes "[Black Market Boss]";
			mes "Do not waste my time.";
			mes "You are not carrying any cards I accept.";
			mes "Cards inserted into equipment don't count.";
			close;
		}

		.@page = 0;
		while (true) {
			.@start = .@page * .PageSize;
			.@end = min(.@start + .PageSize, .@card_count);

			mes "[Black Market Boss]";
			mes "Put a card on the table.";
			mes "^777777Page " + (.@page + 1) + " / " + ((.@card_count + .PageSize - 1) / .PageSize) + "^000000";
			for (.@i = .@start; .@i < .@end; ++.@i)
				mes "^0055FF" + (.@i + 1) + ".^000000 " + mesitemlink(.@card_ids[.@i], false) + " x" + countitem(.@card_ids[.@i]) + " (" + callsub(S_GetOffer, .@card_ids[.@i]) + " CP each)";
			next;

			.@menu$ = "Choose a card.";
			.@option_count = 1;
			.@next_option = 0;
			.@previous_option = 0;
			if (.@end < .@card_count) {
				.@menu$ += ":Next page.";
				.@next_option = ++.@option_count;
			}
			if (.@page > 0) {
				.@menu$ += ":Previous page.";
				.@previous_option = ++.@option_count;
			}
			.@menu$ += ":Cancel.";
			.@cancel_option = ++.@option_count;

			.@action = select(.@menu$);
			if (.@action == .@next_option) {
				++.@page;
				continue;
			}
			if (.@action == .@previous_option) {
				--.@page;
				continue;
			}
			if (.@action == .@cancel_option)
				close;
			break;
		}

		mes "[Black Market Boss]";
		mes "Enter the number shown beside the card.";
		input .@selection, .@start + 1, .@end;
		.@selected_id = .@card_ids[.@selection - 1];
		.@owned = countitem(.@selected_id);

		mes "[Black Market Boss]";
		mes "You have " + .@owned + " " + mesitemlink(.@selected_id, false) + ".";
		mes "How many are you offering me?";
		input .@amount, 1, .@owned;
		.@points_each = callsub(S_GetOffer, .@selected_id);
		.@payment = .@amount * .@points_each;
		.@tier = callsub(S_GetTier, .@selected_id);

		mes "[Black Market Boss]";
		mes "^0055FF" + mesitemlink(.@selected_id, false) + "^000000 x" + .@amount;
		mes "My offer: ^0055FF" + .@payment + " Cash Points^000000.";
		mes "Once they enter my collection, the deal is final.";
		next;
		if (select("Sell the cards.:Keep them.") != 1)
			close;

		// Roll every completed loyalty milestone before committing so the
		// complete reward bundle can be checked for inventory capacity.
		switch (.@tier) {
		case 1:
			.@new_progress = #CC_Normal + .@amount;
			.@milestones = .@new_progress / 30;
			.@remainder = .@new_progress % 30;
			for (.@i = 0; .@i < .@milestones; ++.@i) {
				if (rand(100) < 70)
					.@old_card_albums += 5;
				else
					.@dead_branches += 5;
			}
			break;
		case 2:
			.@new_progress = #CC_Mini + .@amount;
			.@milestones = .@new_progress / 5;
			.@remainder = .@new_progress % 5;
			for (.@i = 0; .@i < .@milestones; ++.@i) {
				if (rand(100) < 50)
					.@magic_card_albums += 1;
				else
					.@dead_branches += 10;
			}
			break;
		case 3:
			.@new_progress = #CC_MVP + .@amount;
			.@milestones = .@new_progress / 5;
			.@remainder = .@new_progress % 5;
			for (.@i = 0; .@i < .@milestones; ++.@i) {
				// 1% album, 40% Bloody Branch x2, 40% Empty Bottle x100, 19% backstab.
				.@roll = rand(100);
				if (.@roll < 1)
					.@mvp_card_albums += 1;
				else if (.@roll < 41)
					.@bloody_branches += 2;
				else if (.@roll < 81)
					.@empty_bottles += 100;
				else
					.@backstabs += 1;
			}
			break;
		default:
			mes "[Black Market Boss]";
			mes "That card no longer belongs in my catalog.";
			mes "Take it away.";
			close;
		}

		.@reward_count = 0;
		if (.@old_card_albums) {
			.@reward_ids[.@reward_count] = 616;
			.@reward_amounts[.@reward_count++] = .@old_card_albums;
		}
		if (.@dead_branches) {
			.@reward_ids[.@reward_count] = 604;
			.@reward_amounts[.@reward_count++] = .@dead_branches;
		}
		if (.@magic_card_albums) {
			.@reward_ids[.@reward_count] = 12246;
			.@reward_amounts[.@reward_count++] = .@magic_card_albums;
		}
		if (.@mvp_card_albums) {
			.@reward_ids[.@reward_count] = 35000;
			.@reward_amounts[.@reward_count++] = .@mvp_card_albums;
		}
		if (.@bloody_branches) {
			.@reward_ids[.@reward_count] = 12103;
			.@reward_amounts[.@reward_count++] = .@bloody_branches;
		}
		if (.@empty_bottles) {
			.@reward_ids[.@reward_count] = 713;
			.@reward_amounts[.@reward_count++] = .@empty_bottles;
		}

		if (.@reward_count && !checkweight2(.@reward_ids, .@reward_amounts)) {
			mes "[Black Market Boss]";
			mes "My payment includes more than your pockets can hold.";
			mes "Make room, then return to my table.";
			close;
		}

		// Revalidate immediately before committing the exchange.
		if (getiteminfo(.@selected_id, ITEMINFO_TYPE) != IT_CARD ||
			getiteminfo(.@selected_id, ITEMINFO_SUBTYPE) != CARD_NORMAL ||
			callsub(S_GetOffer, .@selected_id) != .@points_each ||
			callsub(S_GetTier, .@selected_id) != .@tier ||
			countitem(.@selected_id) < .@amount) {
			mes "[Black Market Boss]";
			mes "Something changed before the deal was closed.";
			mes "I do not conduct uncertain business.";
			close;
		}

		delitem .@selected_id, .@amount;
		#CASHPOINTS += .@payment;
		switch (.@tier) {
		case 1: #CC_Normal = .@remainder; break;
		case 2: #CC_Mini = .@remainder; break;
		case 3: #CC_MVP = .@remainder; break;
		}
		for (.@i = 0; .@i < .@reward_count; ++.@i)
			getitem .@reward_ids[.@i], .@reward_amounts[.@i];
		logmes "Card Collector: sold " + .@amount + " x item " + .@selected_id + " for " + .@payment + " Cash Points; tier " + .@tier + "; milestones " + .@milestones + "; OCA " + .@old_card_albums + "; DB " + .@dead_branches + "; MCA " + .@magic_card_albums + "; MVP albums " + .@mvp_card_albums + "; BDB " + .@bloody_branches + "; bottles " + .@empty_bottles + "; backstabs " + .@backstabs + "; balance " + #CASHPOINTS;

		mes "[Black Market Boss]";
		mes "A clean transaction. I expect nothing less.";
		mes "You received ^0055FF" + .@payment + " Cash Points^000000.";
		mes "Current balance: ^0055FF" + #CASHPOINTS + " Cash Points^000000.";
		if (.@milestones) {
			mes "^FF8800Loyalty reward:^000000";
			if (.@old_card_albums) mes mesitemlink(616, false) + " x" + .@old_card_albums;
			if (.@dead_branches) mes mesitemlink(604, false) + " x" + .@dead_branches;
			if (.@magic_card_albums) mes mesitemlink(12246, false) + " x" + .@magic_card_albums;
			if (.@mvp_card_albums) mes mesitemlink(35000, false) + " x" + .@mvp_card_albums;
			if (.@bloody_branches) mes mesitemlink(12103, false) + " x" + .@bloody_branches;
			if (.@empty_bottles) mes mesitemlink(713, false) + " x" + .@empty_bottles;
		}
		if (.@backstabs) {
			mes "...";
			next;
			mes "[Black Market Boss]";
			mes "You sold the wrong card to the wrong man.";
			mes "A collector should know when not to sell.";
			next;
			close2;
			unitkill getcharid(3);
			end;
		}
		close;

	case 2:
		mes "[Black Market Boss]";
		mes "Your account has ^0055FF" + #CASHPOINTS + " Cash Points^000000.";
		mes "Normal-card loyalty: ^0055FF" + #CC_Normal + " / 30^000000.";
		mes "Miniboss-card loyalty: ^0055FF" + #CC_Mini + " / 5^000000.";
		mes "MVP-card loyalty: ^0055FF" + #CC_MVP + " / 5^000000.";
		close;

	default:
		close;
	}

S_GetOffer:
	.@item_id = getarg(0);
	.@i = inarray(.CardIds[0], .@item_id);
	if (.@i == -1)
		return 0;
	.@lv = getmonsterinfo(.SourceMobIds[.@i], MOB_LV);
	if (.@lv < 1)
		return 0;
	if (inarray(.Offer10[0], .@item_id) != -1)
		return callfunc("F_OfflineCardCollectorOffer", .@lv, 3);
	if (inarray(.MinibossCards[0], .@item_id) != -1)
		return callfunc("F_OfflineCardCollectorOffer", .@lv, 2);
	return callfunc("F_OfflineCardCollectorOffer", .@lv, 1);

S_GetTier:
	.@item_id = getarg(0);
	if (inarray(.Offer10[0], .@item_id) != -1)
		return 3;
	if (inarray(.MinibossCards[0], .@item_id) != -1)
		return 2;
	if (callsub(S_GetOffer, .@item_id) > 0)
		return 1;
	return 0;

OnInit:
	.PageSize = 10;

	freeloop 1;
	.@n = 0;
	.@normal_count = 684;
	for (.@i = 0; .@i < .@normal_count; ++.@i) {
		.CardIds[.@n] = getelementofarray(getvariableofnpc(.CardIds, "card_seller_creation"), .@i);
		.SourceMobIds[.@n] = getelementofarray(getvariableofnpc(.SourceMobIds, "card_seller_creation"), .@i);
		++.@n;
	}
	// Costume stones accepted by the Collector but excluded from the Normal seller.
	.CardIds[.@n] = 6716; .SourceMobIds[.@n] = 3075; ++.@n;
	.CardIds[.@n] = 6717; .SourceMobIds[.@n] = 3075; ++.@n;
	.CardIds[.@n] = 6718; .SourceMobIds[.@n] = 3075; ++.@n;

	.@mini_count = 84;
	for (.@i = 0; .@i < .@mini_count; ++.@i) {
		.CardIds[.@n] = getelementofarray(getvariableofnpc(.CardIds, "miniboss_card_seller_creation"), .@i);
		.SourceMobIds[.@n] = getelementofarray(getvariableofnpc(.SourceMobIds, "miniboss_card_seller_creation"), .@i);
		++.@n;
	}

	// 139 MVP cards. Sources are the lowest-ID MVP dropper of each card.
	setarray .Offer10[0], 4121,4123,4128,4131,4132,4134,4135,4137,4142,4143,4144,4145,4146,4147,4148,4168,4236,4263,4276,4302,4305,4318,4324,4330,4342,4352,4357,4359,4361,4363,4365,4367,4372,4374,4376,4386,4399,4403,4407,4408,4419,4425,4430,4441,4451,4456,4507,4509,4520,4525,4526,4527,4528,4529,4534,4556,4557,4560,4561,4562,4563,4564,4565,4566,4574,4576,4578,4580,4592,4601,4603,4610,4625,4636,4647,4648,4649,4650,4651,4652;
	setarray .Offer10[80], 4662,4671,4672,4673,4674,4675,4676,4677,4678,4679,4680,4681,4682,4683,4697,27020,27025,27081,27104,27106,27113,27119,27126,27150,27151,27152,27162,27164,27180,27182,27265,27294,27298,27305,27318,27319,27321,27326,27327,27346,27362,27363,31023,31026,300007,300078,300079,300080,300099,300100,300107,300145,300151,300217,300227,300239,300248,300262,300281;

	setarray .MvpSources[80], 3505,3242,3243,3245,3220,3221,3222,3223,3224,3225,3240,3241,3244,3246,3124,3254,3658,3758,3757,3073,3796,3804,2068,3633,3628,3621,1885,2529,3741,3181,2942,20273,20277,20340,20346,20381,3097,3000,3029,20419,20421,20422,2996,3473,20601,20642,20620,20621,20667,20668,20648,20843,20811,21301,21314,21395,20943,20928,20934;

	.@mvp_count = getarraysize(.Offer10);

		debugmes "Card Collector: Normal catalog copy did not start at Poring.";

	copyarray $@CC_MvpAlbumCards[0], .Offer10[0], getarraysize(.Offer10);

function	script	F_OfflineMvpCardAlbum	{
	.@size = getarraysize($@CC_MvpAlbumCards);
	if (.@size <= 0) {
		displaymessage strcharinfo(0), "The MVP Card Album is not ready yet.";
		getitem 35000,1;
		return;
	}
	getitem $@CC_MvpAlbumCards[rand(.@size)],1;
	return;
}

// Bodyguards: visible and proximity-reactive, but they do not open dialogue.
paramk,41,161,4	script	Guard#card_collector_1	405,2,1,{
	callsub S_Warn;
	end;

OnTouch:
	callsub S_Warn;
	end;

S_Warn:
	if (@CardGuardWarningUntil > gettimetick(2))
		return;
	@CardGuardWarningUntil = gettimetick(2) + 3;

	switch (rand(3)) {
	case 0:
		npctalk "Easy. Don't try anything.", strnpcinfo(0), bc_self;
		break;
	case 1:
		npctalk "Watch yourself. The Boss is protected.", strnpcinfo(0), bc_self;
		break;
	default:
		npctalk "Keep your hands where I can see them.", strnpcinfo(0), bc_self;
		break;
	}
	return;
}

paramk,41,157,2	duplicate(Guard#card_collector_1)	Guard#card_collector_2	405,2,1
