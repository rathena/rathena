//===== rAthena Script =======================================
//= Transcended RO - Item ID Checker
//===== Description: =========================================
//= Public NPC at sec_in02,142,156 and izlude,161,170.
//= Drop an item into the sell window and press Sell.
//= The NPC reports the ItemID / name / amount. The item is kept.
//= Useful for finding IDs of items that have no iteminfo yet.
//= GM command: @itemcheck (GM level 99+) opens the same window
//= from anywhere.
//============================================================

sec_in02,142,156,4	script	Item Checker#gm	4_M_SOULREAPER,{
	set @itemchk_remote, 0;
	mes "[Item Checker]";
	mes "Drop the item into the sell window and press ^0055FFSell^000000.";
	mes "I will tell you its ItemID.";
	close2;
	callshop "gm_itemcheck_shop", 2;
	npcshopattach "gm_itemcheck_shop";
	end;

// @itemcheck is used far from the NPC -> no mes/close dialog
// (npc_checknear would fail and hang the script). Use chat output.
OnAtItemCheck:
	set @itemchk_remote, 1;
	dispbottom "[Item Checker] Drop the item into the sell window and press Sell.";
	callshop "gm_itemcheck_shop", 2;
	npcshopattach "gm_itemcheck_shop";
	end;

OnInit:
	initnpctimer;
	// Bind only once (duplicates also run OnInit).
	// @itemcheck — GM level 99+ only (no # charcommand)
	if (strnpcinfo(2) == "gm")
		bindatcmd "itemcheck", strnpcinfo(3) + "::OnAtItemCheck", 99, 100;
	end;

OnTimer1000:
    showscript("Apple Checker!!!");
    setnpctimer 0;
    end;

OnSellItem:
	if (@itemchk_remote) {
		for (set .@i, 0; .@i < getarraysize(@sold_nameid); set .@i, .@i + 1)
			dispbottom "[Item Checker] ItemID: " + @sold_nameid[.@i] + " x" + @sold_quantity[.@i] + " - " + getitemname(@sold_nameid[.@i]), 0x00FF00;
		end;
	}
	mes "[Item Checker]";
	mes "====================";
	set .@n, getarraysize(@sold_nameid);
	for (set .@i, 0; .@i < .@n; set .@i, .@i + 1) {
		set .@id, @sold_nameid[.@i];
		set .@qty, @sold_quantity[.@i];
		set .@ids[.@i], .@id;
		mes "ItemID: ^00AA00" + .@id + "^000000  x" + .@qty;
		mes "Name: " + getitemname(.@id);
		mes "====================";
	}
	next;
	if (select("Report to GM (make ItemInfo)", "Close") == 2)
		close;

	// Report -> SQL table itemcheck_report.
	// tools/discord_item_report.php relays new rows to Discord.
	// One report per ItemID (UNIQUE nameid) to stop spam.
	mes "[Item Checker]";
	for (set .@i, 0; .@i < .@n; set .@i, .@i + 1) {
		set .@id, .@ids[.@i];
		set .@dup, 0;
		query_sql("SELECT COUNT(*) FROM `itemcheck_report` WHERE `nameid` = " + .@id, .@dup);
		if (.@dup) {
			mes "ItemID ^00AA00" + .@id + "^000000 was already reported.";
			continue;
		}
		query_sql("INSERT INTO `itemcheck_report` (`nameid`, `item_name`, `account_id`, `char_id`, `char_name`, `report_time`) VALUES ("
			+ .@id + ", '" + escape_sql(getitemname(.@id)) + "', " + getcharid(3) + ", " + getcharid(0) + ", '"
			+ escape_sql(strcharinfo(0)) + "', NOW())");
		mes "ItemID ^00AA00" + .@id + "^000000 reported. Thank you!";
	}
	close;
}
