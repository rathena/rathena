moc_para01,53,37,4	script	Eat Gear Mimic	464,{
    mes "[Eat Gear Mimic]";
    
    // 1. Check if player has joined the Eden Group
    if (!callfunc(\"F_HasEdenGroupMark\")) {
        mes "Grrr... You smell like an outsider!";
        mes "I only deal with registered members of the ^0000FFEden Group^000000!";
        close;
    }

    mes "Give me your refined gear, crunch, crunch...";
    mes "I eat Level 3 or 4 weapons, armors, shields, garments, and shoes!";
    next;

    if (select("Feed a gear:Cancel") == 2) {
        mes "[Eat Gear Mimic]";
        mes "Picky human... come back when you have tasty gear.";
        close;
    }

    // Define the blacklist
    setarray .blacklist, 
        2301, 2302, 2303, 2304, 2305, 2307, 2308, 2309, 2310, 2311, 
        2312, 2314, 2316, 2318, 2319, 2320, 2321, 2324, 2326, 2252, 
        2101, 2103, 2105, 2107, 2109, 2111, 2113, 2114, 2116, 2117,
        2119, 2121, 2123, 2125, 2127, 2129, 2131, 2133, 2135, 2137,
        2139, 2141, 2143, 2145, 2147, 2149, 2151, 2153, 2155, 2157,
        2159, 2161, 2163, 2165, 2167, 2169, 2171, 2173, 2175, 2177,
        2181, 2183, 2185, 2187, 2189, 2191, 2193, 2195, 2197, 2199;

    // Retrieve full character inventory data mapping
    getinventorylist;
    .@eligible_count = 0;
    clearinit;

    // Gather qualifying equipment profiles
    for (.@i = 0; .@i < @inventorylist_count; .@i++) {
        .@item_id = @inventorylist_id[.@i];
        .@refine  = @inventorylist_refine[.@i];
        .@type    = @inventorylist_type[.@i];

        // Filter out non-equips, zero refines, or broken status variables
        if (.@refine < 7 || @inventorylist_isbroken[.@i]) continue;
        if (.@type != IT_WEAPON && .@type != IT_ARMOR) continue;

        // Run checking arrays against safety blacklist definitions
        .@is_blacklisted = 0;
        for (.@j = 0; .@j < getarraysize(.blacklist); .@j++) {
            if (.@item_id == .blacklist[.@j]) {
                .@is_blacklisted = 1;
                break;
            }
        }
        if (.@is_blacklisted) continue;

        // Custom validation check rules for item-database weapon tier lines
        if (.@type == IT_WEAPON) {
            .@w_lv = getiteminfo(.@item_id, ITEMINFO_ELV);
            if (.@w_lv < 3) continue;
        }

        // Map selection options array
        .@menu_display$[.@eligible_count] = "[Slot " + (.@i + 1) + "] +" + .@refine + " " + getitemname(.@item_id);
        .@inventory_indices[.@eligible_count] = .@i;
        .@eligible_count++;
    }

    if (.@eligible_count == 0) {
        mes "[Eat Gear Mimic]";
        mes "Bah! You don't have anything appetizing.";
        mes "Bring me ^FF0000+7 or higher Level 3/4 weapons, or highly refined armors^000000 that aren't broken!";
        close;
    }

    mes "[Eat Gear Mimic]";
    mes "Oho! Which gear should I snack on?";
    mes "^666666(Warning: The mimic will eat your gear, but only has a 25% chance to spit out a reward!)^000000";
    next;

    // Render selection target mapping
    .@selection$ = "";
    for (.@i = 0; .@i < .@eligible_count; .@i++) {
        .@selection$ = .@selection$ + .@menu_display$[.@i] + ":";
    }
    .@selection$ = .@selection$ + "Cancel";

    .@chosen = select(.@selection$) - 1;
    if (.@chosen == .@eligible_count) {
        mes "[Eat Gear Mimic]";
        mes "Hmph. Changed your mind?";
        close;
    }

    // Assign internal validation indices for execution logic
    .@selected_index  = .@inventory_indices[.@chosen];
    .@selected_item   = @inventorylist_id[.@selected_index];
    .@selected_refine = @inventorylist_refine[.@selected_index];
    .@selected_type   = @inventorylist_type[.@selected_index];

    mes "[Eat Gear Mimic]";
    mes "Are you sure you want me to eat your ^FF0000(+" + .@selected_refine + " " + getitemname(.@selected_item) + ")^000000?";
    next;

    if (select("Yes, feed him:No, save it") == 2) {
        mes "[Eat Gear Mimic]";
        mes "Smart choice... almost lost a good tool.";
        close;
    }

    // Re-verify that the item slot index matches our stored target item ID
    getinventorylist;
    if (@inventorylist_id[.@selected_index] == .@selected_item && @inventorylist_refine[.@selected_index] == .@selected_refine) {
        
        // Remove item from inventory via index slot mapping (Gear is lost here!)
        delitemidx(@inventorylist_idx[.@selected_index], 1);

        mes "[Eat Gear Mimic]";
        mes "*CRUNCH* *MUNCH* ... Gulp.";
        next;

        // 25% Chance Validation Engine
        if (rand(1, 100) <= 20) {
            // SUCCESS LOGIC (20% Rate)
            if (.@selected_type == IT_WEAPON) {
                // Weapon Rewards
                if (.@selected_refine == 7)       getitem 6230, 1;
                else if (.@selected_refine == 8)  getitem 6230, 2;
                else if (.@selected_refine >= 9)  getitem 6230, 5;
            } else {
                // Armor / Shield / Garment / Shoe Rewards
                if (.@selected_refine == 7)       getitem 6231, 1;
                else if (.@selected_refine == 8)  getitem 6231, 2;
                else if (.@selected_refine >= 9)  getitem 6231, 5;
            }

            specialeffect EF_ORANGESPARK;
            soundeffect "item_store.wav", 0;

            mes "[Eat Gear Mimic]";
            mes "Puuuh!! That tasted awesome! Here, take this something extra I found in my teeth!";
            close;
        } else {
            // FAILURE LOGIC (80% Rate)
            specialeffect EF_BOMBER;

            mes "[Eat Gear Mimic]";
            mes "*BURP* ... Ah, that hit the spot! Too bad it completely dissolved in my stomach acid. Thanks for the meal, adventurer!";
            close;
        }
    } else {
        // Fallback error validation loop handler
        mes "[Eat Gear Mimic]";
        mes "Hey... your inventory inventory layout shifted! Stop messing around.";
        close;
    }
}
