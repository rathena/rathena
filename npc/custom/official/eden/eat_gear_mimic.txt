moc_para01,53,37,4	script	Eat Gear Mimic	464,{
    mes "[Eat Gear Mimic]";
    
    // 1. Check if player has joined the Eden Group
    if (!callfunc("F_HasEdenGroupMark")) {
        mes "Grrr... You smell like an outsider!";
        mes "I only deal with registered members of the ^0000FFEden Group^000000!";
        close;
    }

    mes "Give me your refined gear, crunch, crunch...";
    mes "I eat Level 3 or 4 weapons, armors, shields, garments, and shoes!";
    next;

    if (select("Feed a gear:Cancel") == 2) {
        mes "[Eat Gear Mimic]";
        mes "Picky human... come back when you have tasty gear.";
        close;
    }

    // Define the blacklist
    setarray .blacklist, 
        2301, 2302, 2303, 2304, 2305, 2307, 2308, 2309, 2310, 2311, 
        2312, 2314, 2316, 2318, 2319, 2320, 2321, 2324, 2326, 2252, 
        2101, 2103, 2105, 2187, 2189, 2201, 2203, 2205, 2212, 2218, 
        2506, 5054, 5087, 5107;

    // Clear tracking arrays safely
    cleararray .@items, 0, 128;
    cleararray .@refines, 0, 128;
    cleararray .@indices, 0, 128;
    cleararray .@types, 0, 128;
    .@menus$ = "";
    .@count = 0;

    // Populates native character temporary inventory arrays
    getinventorylist;

    // Bypass loop timeout engine limits safely
    freeloop(1);

    // Loop through player inventory
    for (.@i = 0; .@i < @inventorylist_count; .@i++) {
        .@item_id = @inventorylist_id[.@i];
        .@refine  = @inventorylist_refine[.@i];
        .@type    = getiteminfo(.@item_id, ITEMINFO_TYPE);

        // 1. Safety: Do not process items that are actively equipped by the player
        if (@inventorylist_equip[.@i]) continue;

        // 2. Custom Rule: Only accept gear with refines (+7 or higher)
        if (.@refine < 7) continue;

        // 3. Simple Broad Category Check: Must be standard armor or weapon
        if (.@type != IT_ARMOR && .@type != IT_WEAPON) continue;

        // 4. Blacklist Array Cross-Check
        .@is_blacklisted = 0;
        for (.@b = 0; .@b < 34; .@b++) { 
            if (.@item_id == .blacklist[.@b]) {
                .@is_blacklisted = 1;
                break;
            }
        }
        if (.@is_blacklisted) continue; 

        // 5. Exclude Upper/Middle/Lower Headgears cleanly via common mask evaluation
        .@loc = getiteminfo(.@item_id, ITEMINFO_LOC);
        if (.@type == IT_ARMOR && (.@loc & EQP_HEAD_TOP || .@loc & EQP_HEAD_MID || .@loc & EQP_HEAD_LOW)) continue;

        // Store selected variables safely inside local index pointers
        .@items[.@count]   = .@item_id;
        .@refines[.@count] = .@refine;
        .@indices[.@count] = .@i;        
        .@types[.@count]   = .@type;      

        // Build the option choice loop window sequence cleanly without leaving a trailing separator
        if (.@menus$ == "") {
            .@menus$ = getitemname(.@item_id) + " (+" + .@refine + ")";
        } else {
            .@menus$ = .@menus$ + ":" + getitemname(.@item_id) + " (+" + .@refine + ")";
        }
        .@count++;
    }

    // Turn safety guards back on
    freeloop(0);

    // Safely exit if no eligible items passed all requirements
    if (.@count == 0) {
        mes "[Eat Gear Mimic]";
        mes "You don't have any suitable, unequipped gear (+7 or higher) in your bag!";
        close;
    }

    // Process user selection index cleanly from choices
    .@choice = select(.@menus$);
    .@selected_item   = .@items[.@choice - 1];
    .@selected_refine = .@refines[.@choice - 1];
    .@selected_index  = .@indices[.@choice - 1];
    .@selected_type    = .@types[.@choice - 1];

    mes "[Eat Gear Mimic]";
    mes "Are you sure you want me to eat your ^FF0000(+" + .@selected_refine + " " + getitemname(.@selected_item) + ")^000000?";
    next;

    if (select("Yes, feed him:No, save it") == 2) {
        mes "[Eat Gear Mimic]";
        mes "Smart choice... almost lost a good tool.";
        close;
    }

    // Re-verify that the item slot index matches our stored target item ID
    getinventorylist;
    if (@inventorylist_id[.@selected_index] == .@selected_item && @inventorylist_refine[.@selected_index] == .@selected_refine) {
        
        // Remove item from inventory via index slot mapping
        delitemidx(@inventorylist_idx[.@selected_index], 1);

        mes "[Eat Gear Mimic]";
        mes "*CRUNCH* *MUNCH* ... Delicious!";

        // Reward Logic engine
        if (.@selected_type == IT_WEAPON) {
            // Weapon Rewards
            if (.@selected_refine == 7)       getitem 6230, 1;
            else if (.@selected_refine == 8)  getitem 6230, 2;
            else if (.@selected_refine >= 9)  getitem 6230, 5;
        } else {
            // Armor / Shield / Garment / Shoe Rewards
            if (.@selected_refine == 7)       getitem 6234, 1;
            else if (.@selected_refine == 8)  getitem 6234, 2;
            else if (.@selected_refine >= 9)  getitem 6234, 5;
        }

    } else {
        mes "[Eat Gear Mimic]";
        mes "Hey! Stop moving things around in your bag! The item changed positions!";
    }
    close;
}
